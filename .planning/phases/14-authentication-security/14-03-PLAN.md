2. Create `frontend/src/lib/auth/api.ts` with auth API functions:
   - `loginApi(email, password)` — POST `/api/auth/login` with `credentials: 'include'`, returns `{ access_token }`. Content-Type `application/json`.
   - `registerApi(data: RegisterData)` — POST `/api/auth/register` with `credentials: 'include'`, returns `{ access_token }`.
   - `refreshApi()` — POST `/api/auth/refresh` with `credentials: 'include'` (no body needed, cookie is sent automatically), returns `{ access_token }`. On failure throws error.
   - `logoutApi()` — POST `/api/auth/logout` with `credentials: 'include'` (no body needed).
   - `fetchUserApi(accessToken: string)` — GET `/api/auth/me` with `Authorization: Bearer ${accessToken}` header, returns User object.

   CRITICAL: ALL fetch calls MUST include `credentials: 'include'` for the httpOnly cookie to be sent by the browser. Without this, the cookie is not attached to requests.

   All functions should throw on non-OK responses. Parse error detail from response JSON when available:
   ```typescript
   if (!response.ok) {
     const error = await response.json().catch(() => ({ detail: 'Request failed' }))
     throw new Error(error.detail || 'Request failed')
   }
   ```

2. Create `frontend/src/lib/auth/AuthContext.tsx`:
   ```typescript
   import { createContext, useState, useEffect, useCallback } from 'react'
   import type { AuthContextValue, User, RegisterData } from './types'
   import { loginApi, registerApi, refreshApi, logoutApi, fetchUserApi } from './api'

   export const AuthContext = createContext<AuthContextValue | null>(null)

   export function AuthProvider({ children }: { children: React.ReactNode }) {
     const [user, setUser] = useState<User | null>(null)
     const [accessToken, setAccessToken] = useState<string | null>(null)
     const [isLoading, setIsLoading] = useState(true)

     // Silent refresh on mount
     useEffect(() => {
       refreshApi()
         .then(async (data) => {
           setAccessToken(data.access_token)
           const userData = await fetchUserApi(data.access_token)
           setUser(userData)
         })
         .catch(() => {
           // Refresh failed — user is not authenticated
           setUser(null)
           setAccessToken(null)
         })
         .finally(() => setIsLoading(false))
     }, [])

     const login = useCallback(async (email: string, password: string) => {
       const data = await loginApi(email, password)
       setAccessToken(data.access_token)
       const userData = await fetchUserApi(data.access_token)
       setUser(userData)
     }, [])

     const register = useCallback(async (registerData: RegisterData) => {
       const data = await registerApi(registerData)
       setAccessToken(data.access_token)
       const userData = await fetchUserApi(data.access_token)
       setUser(userData)
     }, [])

     const logout = useCallback(async () => {
       await logoutApi().catch(() => {
         // Ignore logout API errors — clear local state regardless
       })
       setAccessToken(null)
       setUser(null)
     }, [])

     return (
       <AuthContext.Provider
         value={{
           user,
           isLoading,
           isAuthenticated: !!user,
           accessToken,
           login,
           register,
           logout,
         }}
       >
         {children}
       </AuthContext.Provider>
     )
   }
   ```

   IMPORTANT per user decision: On refresh token expiry, silently redirect to /auth with NO error toast. The redirect happens naturally via the beforeLoad guard when isAuthenticated becomes false.
