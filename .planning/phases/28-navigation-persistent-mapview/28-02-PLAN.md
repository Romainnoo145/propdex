**Step 1: Create `frontend/src/components/layout/InspectorPanel.tsx`**

Build a right-edge slide-over panel that displays parcel details from scan results. Key design decisions from research:
- No backdrop (map stays fully interactive underneath)
- CSS `translate-x` animation for slide-in/out
- Uses existing z-index variable `--z-property-panel` (300)
- Escape key to close (reuse existing `useEscapeKey` hook)
- Width: `w-80` (320px) with `max-w-[90vw]` for small screens

```typescript
import { useEscapeKey } from '@/hooks/useEscapeKey'
import { X } from 'lucide-react'
import type { ScanResultItem } from '@/lib/types/scan'

interface InspectorPanelProps {
  isOpen: boolean
  onClose: () => void
  parcel: ScanResultItem | null
}

export function InspectorPanel({ isOpen, onClose, parcel }: InspectorPanelProps) {
  useEscapeKey(onClose, isOpen)

  return (
    <div
      className={`absolute right-0 top-0 h-full w-80 max-w-[90vw]
                  bg-grey-900/90 backdrop-blur-xl border-l border-white/10
                  shadow-lg flex flex-col
                  transition-transform duration-300 ease-out
                  ${isOpen ? 'translate-x-0' : 'translate-x-full'}`}
      style={{ zIndex: 'var(--z-property-panel)' }}
      aria-hidden={!isOpen}
    >
      {/* Header */}
      <div className="flex items-center justify-between px-4 py-3 border-b border-white/10 shrink-0">
        <span className="text-sm font-semibold text-grey-100">Perceeldetails</span>
        <button
          onClick={onClose}
          className="text-grey-400 hover:text-white transition-colors p-1 rounded hover:bg-white/10"
          aria-label="Sluiten"
        >
          <X size={16} />
        </button>
      </div>

      {/* Content */}
      <div className="flex-1 overflow-y-auto p-4">
        {parcel ? (
          <div className="space-y-4">
            {/* Parcel ID */}
            <div>
              <p className="text-xs font-medium text-grey-400 uppercase tracking-wider">Perceel ID</p>
              <p className="text-sm text-grey-100 font-mono mt-0.5">{parcel.parcel_id}</p>
            </div>

            {/* Address */}
            {parcel.address && (
              <div>
                <p className="text-xs font-medium text-grey-400 uppercase tracking-wider">Adres</p>
                <p className="text-sm text-grey-100 mt-0.5">{parcel.address}</p>
              </div>
            )}

            {/* Bestemming */}
            {parcel.bestemming && (
              <div>
                <p className="text-xs font-medium text-grey-400 uppercase tracking-wider">Bestemming</p>
                <span className="inline-block mt-1 px-2 py-0.5 bg-amber-accent/20 text-amber-accent text-xs font-medium rounded">
                  {parcel.bestemming}
                </span>
              </div>
            )}

            {/* Area */}
            {parcel.area_m2 != null && (
              <div>
                <p className="text-xs font-medium text-grey-400 uppercase tracking-wider">Oppervlakte</p>
                <p className="text-sm text-grey-100 mt-0.5">
                  {parcel.area_m2.toLocaleString('nl-NL')} m²
                </p>
              </div>
            )}

            {/* Has building */}
            <div>
              <p className="text-xs font-medium text-grey-400 uppercase tracking-wider">Bebouwing</p>
              <p className="text-sm text-grey-100 mt-0.5">
                {parcel.has_pand ? 'Ja — pand aanwezig' : 'Nee — onbebouwd'}
              </p>
            </div>

            {/* Score */}
            {parcel.opportunity_score != null && (
              <div>
                <p className="text-xs font-medium text-grey-400 uppercase tracking-wider">Kanscore</p>
                <div className="flex items-center gap-2 mt-1">
                  <div className="flex-1 h-2 bg-grey-800 rounded-full overflow-hidden">
                    <div
                      className="h-full bg-amber-accent rounded-full transition-all"
                      style={{ width: `${Math.round(parcel.opportunity_score * 100)}%` }}
                    />
                  </div>
                  <span className="text-sm font-semibold text-grey-100 tabular-nums">
                    {Math.round(parcel.opportunity_score * 100)}
                  </span>
                </div>
              </div>
            )}

            {/* Qualification */}
            {parcel.qualification && (
              <div>
                <p className="text-xs font-medium text-grey-400 uppercase tracking-wider">Kwalificatie</p>
                <p className="text-sm text-grey-100 mt-0.5 capitalize">{parcel.qualification}</p>
              </div>
            )}

            {/* Recent permits count */}
            {parcel.recent_permits != null && (
              <div>
                <p className="text-xs font-medium text-grey-400 uppercase tracking-wider">Recente vergunningen</p>
                <p className="text-sm text-grey-100 mt-0.5">{parcel.recent_permits}</p>
              </div>
            )}
          </div>
        ) : (
          <div className="flex items-center justify-center h-32">
            <p className="text-sm text-grey-500">Selecteer een perceel op de kaart</p>
          </div>
        )}
      </div>
    </div>
  )
}
```

Note: Use design token colors from Phase 27 (grey-100, grey-400, grey-800, grey-900, amber-accent) instead of hardcoded colors. Check `ScanResultItem` type for available fields:
- Look up `frontend/src/lib/types/scan.ts` to confirm the fields on `ScanResultItem` before implementing. Adjust the displayed fields to match what actually exists on the type.

   ```typescript
   // Open inspector on parcel click (when score layer exists)
   useEffect(() => {
     if (!mapReady || !mapRef.current) return
     const map = mapRef.current
     if (!map.getSource('parcel-scores')) return

     const handleClick = (e: maplibregl.MapLayerMouseEvent) => {
       if (!e.features?.length) return
       const feature = e.features[0]
       const props = feature?.properties
       if (!props) return

       // Build a partial ScanResultItem from feature properties
       const parcel: ScanResultItem = {
         parcel_id: props.parcel_id as string,
         address: props.address as string | undefined,
         bestemming: props.bestemming as string | undefined,
         area_m2: props.area_m2 as number | undefined,
         has_pand: props.has_pand as boolean | undefined,
         opportunity_score: props.opportunity_score as number | undefined,
         qualification: props.qualification as string | undefined,
         recent_permits: props.recent_permits as number | undefined,
       }
       setInspectorParcel(parcel)
     }

     map.on('click', 'parcel-scores-fill', handleClick)
     return () => {
       map.off('click', 'parcel-scores-fill', handleClick)
     }
   }, [mapReady, mapRef])
   ```
