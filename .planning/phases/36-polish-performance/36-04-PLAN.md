    **SideRailNav.stories.tsx:**
    ```tsx
    import type { Meta, StoryObj } from '@storybook/react-vite';
    import { createRouter, RouterProvider, createMemoryHistory } from '@tanstack/react-router';
    import { routeTree } from '@/routeTree.gen';
    import SideRailNav from '@/components/layout/SideRailNav';

    const makeRouter = (initialPath: string) =>
      createRouter({
        routeTree,
        history: createMemoryHistory({ initialEntries: [initialPath] }),
      });

    const meta: Meta<typeof SideRailNav> = {
      title: 'Layout/SideRailNav',
      component: SideRailNav,
      parameters: { layout: 'fullscreen' },
    };
    export default meta;
    type Story = StoryObj<typeof SideRailNav>;

    // Wrap with router + dark shell that matches production layout
    const withRouter = (path: string) =>
      (Story: React.ComponentType) => (
        <RouterProvider router={makeRouter(path)}>
          <div className="flex h-screen bg-grey-950">
            <Story />
            <div className="flex-1 bg-grey-900 flex items-center justify-center">
              <span className="text-grey-400 text-sm">Kaartinhoud</span>
            </div>
          </div>
        </RouterProvider>
      );

    export const MapActive: Story = {
      decorators: [withRouter('/map')],
    };

    export const ResultatenActive: Story = {
      decorators: [withRouter('/resultaten')],
    };

    export const ProjectenActive: Story = {
      decorators: [withRouter('/projecten')],
    };
    ```

    Read SideRailNav.tsx first and adjust any required props (e.g., if it needs `sidebarOpen`/`onToggleSidebar` props, add those as Story args). Match actual component prop interface.
