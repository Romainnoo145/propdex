<objective>
Add retry logic to BAG WFS page fetching and coverage validation to the scan pipeline, so that BAG failures are retried instead of silently producing garbage data, and scan operators get clear warnings about suspiciously low coverage.

Purpose: The scan pipeline currently catches BAG WFS exceptions and continues with `bag_features = []`, which means every parcel gets `has_pand=false`, no bouwjaar, no addresses, and wrong scoring. This is the root cause of garbage scan data. Fix it by retrying and aborting on persistent failure, then validate coverage ratios post-scan.

Output: Updated BAG WFS client with per-page retry, updated area_scan_job that aborts on BAG failure and computes coverage stats, Alembic migration for new AreaScan fields, unit tests.
</objective>

    ```python
    # ── Coverage validation ──
    parcels_with_buildings = sum(1 for p, *_ in scored_parcels if p.has_pand)
    total_parcels = len(scored_parcels)
    coverage_ratio = parcels_with_buildings / total_parcels if total_parcels > 0 else 0.0

    # Address resolution: of parcels with buildings, how many got an address?
    parcels_with_address = sum(
        1 for p, *_ in scored_parcels if p.has_pand and p.address
    )
    address_resolution_ratio = (
        parcels_with_address / parcels_with_buildings
        if parcels_with_buildings > 0
        else 0.0
    )

    scan.coverage_ratio = coverage_ratio
    scan.address_resolution_ratio = address_resolution_ratio

    # Warn on suspiciously low building coverage (>90% has_pand=false)
    if total_parcels > 0 and coverage_ratio < 0.10:
        logger.warning(
            "Scan %s [%s]: suspiciously low building coverage: %.1f%% "
            "(%d/%d parcels have buildings). BAG data may be incomplete.",
            scan_id, scan.wijk_naam or "unknown",
            coverage_ratio * 100, parcels_with_buildings, total_parcels,
        )

    # Warn on low address resolution (>50% of building-parcels unresolved)
    if parcels_with_buildings > 0 and address_resolution_ratio < 0.50:
        logger.warning(
            "Scan %s [%s]: low address resolution: %.1f%% "
            "(%d/%d parcels with buildings have addresses).",
            scan_id, scan.wijk_naam or "unknown",
            address_resolution_ratio * 100, parcels_with_address, parcels_with_buildings,
        )

    # Set status to completed_with_warnings if thresholds breached
    has_coverage_warning = total_parcels > 0 and coverage_ratio < 0.10

    ```

    @pytest.mark.asyncio
    class TestRetryWithBackoffErrors:
        """Tests for retry_with_backoff with retry_on_errors=True."""

        async def test_retries_on_5xx_and_succeeds(self):
            """Returns successful response after 5xx failures."""
            fail_resp = MagicMock(spec=httpx.Response, status_code=500)
            ok_resp = MagicMock(spec=httpx.Response, status_code=200)
            fn = AsyncMock(side_effect=[fail_resp, fail_resp, ok_resp])

            with patch("asyncio.sleep", new_callable=AsyncMock):
                result = await retry_with_backoff(fn, "test", retries=3, retry_on_errors=True)

            assert result.status_code == 200
            assert fn.call_count == 3

        async def test_retries_on_request_error_and_succeeds(self):
            """Recovers from network error on retry."""
            ok_resp = MagicMock(spec=httpx.Response, status_code=200)
            fn = AsyncMock(side_effect=[httpx.RequestError("timeout"), ok_resp])

            with patch("asyncio.sleep", new_callable=AsyncMock):
                result = await retry_with_backoff(fn, "test", retries=3, retry_on_errors=True)

            assert result.status_code == 200
            assert fn.call_count == 2

        async def test_raises_after_all_retries_exhausted(self):
            """All retries fail — last exception propagates."""
            fn = AsyncMock(side_effect=httpx.RequestError("timeout"))

            with patch("asyncio.sleep", new_callable=AsyncMock):
                with pytest.raises(httpx.RequestError):
                    await retry_with_backoff(fn, "test", retries=3, retry_on_errors=True)

            """Successful first call has no retries."""
            ok_resp = MagicMock(spec=httpx.Response, status_code=200)

            assert fn.call_count == 1
