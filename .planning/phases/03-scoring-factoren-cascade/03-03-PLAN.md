```python
# M050/S03 D3 — sparse-data runtime redistribution for 3 new BRK risico factors.
# When any of {hypotheek_risico, eigendoms_complexiteit, koopwaarde_trend} is
# missing (brk_records=[] or factor-specific input None), exclude its
# default_weight from the risico-category total (100) and rescale gewicht +
# bijdrage of non-missing risico factors proportionally so they sum back to 100.
# Missing BRK factors stay in breakdown with missing=True (frontend cards can
# show 'BRK-data nodig'); their bijdrage stays 0 — only non-missing risico
# factors receive the rescale multiplier.
missing_brk_keys = {
    k for k in _SPARSE_REDISTRIBUTABLE_RISICO
    if k in breakdown and breakdown[k].missing
}
if missing_brk_keys:
    missing_weight = sum(
        FACTOR_REGISTRY[k].default_weight for k in missing_brk_keys
    )
    active_weight = 100 - missing_weight
    if active_weight > 0:
        scale = 100.0 / active_weight
        for k, factor in breakdown.items():
            if (
                factor.categorie == "risico"
                and k not in missing_brk_keys
                and not factor.missing
            ):
                factor.gewicht = round(factor.gewicht * scale, 4)
                factor.bijdrage = round(
                    factor.genormaliseerde_score * factor.gewicht, 4
                )
                factor.percentage_of_category = round(factor.gewicht / 100.0, 4)
```

async def rescore_parcel(
    ctx: dict[str, Any],
    *,
    organization_id: str,    # UUID stringified for ARQ JSON-serialization
    parcel_id: str,
) -> dict[str, Any]:
    """ARQ job — re-compute Score v3 for a single (org, parcel) pair.

    Triggered after BRK persist (D7) via enqueue_rescore_parcel. Idempotent via
    ARQ _job_id='rescore:{org}:{parcel}' — concurrent enqueues collapse.

    Steps:
        1. Load latest ScanResult for parcel via DB session.
        2. Call enrich_parcel_data_with_brk(flags, org_uuid, db) — D1+D8+D9.
        3. Call compute_score_v3(flags) — D3 redistribution runs internally.
        4. Write projected score_breakdown back into scan_result.flags +
           bump flags['m050_s03_factors_at'] = datetime.utcnow().isoformat().
        5. Commit + return summary dict for ARQ result-store.

    Returns:
        {'parcel_id', 'updated': bool, 'breakdown_keys': list[str],
         'm050_s03_factors_at': str, 'duration_ms': int}
    """
    from app.models.scan_result import ScanResult
    from app.services.scoring_pipeline import enrich_parcel_data_with_brk
    from app.services.score_v3 import compute_score_v3

    org_uuid = uuid.UUID(organization_id)
    engine = create_async_engine(settings.DATABASE_URL)
    async_session_factory = async_sessionmaker(engine, expire_on_commit=False)

    started_at = time.time()
    async with async_session_factory() as db:
        stmt = (
            select(ScanResult)
            .where(ScanResult.parcel_id == parcel_id)
            .order_by(ScanResult.created_at.desc())
            .limit(1)
        )
        result = await db.execute(stmt)
        scan = result.scalar_one_or_none()
        if scan is None:
            logger.warning(
                "rescore_parcel: no scan_result for parcel=%s org=%s",
                parcel_id, organization_id,
            )
            await engine.dispose()
            return {"parcel_id": parcel_id, "updated": False, "reason": "no_scan_result"}

        flags = dict(scan.flags or {})
        flags["parcel_id"] = parcel_id
        flags = await enrich_parcel_data_with_brk(flags, org_uuid, db)
        score_result = compute_score_v3(flags)
        flags["score_breakdown"] = {
            k: {
                "genormaliseerde_score": v.genormaliseerde_score,
                "gewicht": v.gewicht,
                "bijdrage": v.bijdrage,
                "categorie": v.categorie,
                "toelichting_kort": v.toelichting_kort,
                "missing": v.missing,
            }
            for k, v in score_result.score_breakdown.items()
        }
        flags["m050_s03_factors_at"] = datetime.utcnow().isoformat()
        scan.flags = flags
        await db.commit()

    await engine.dispose()
    return {
        "parcel_id": parcel_id,
        "updated": True,
        "breakdown_keys": list(score_result.score_breakdown.keys()),
        "m050_s03_factors_at": flags["m050_s03_factors_at"],
        "duration_ms": int((time.time() - started_at) * 1000),
    }


async def enqueue_rescore_parcel(
    arq_pool: Any,           # arq.connections.ArqRedis
    organization_id: uuid.UUID,
    parcel_id: str,
) -> str | None:
    """Enqueue rescore_parcel via ARQ with deterministic _job_id for F7 dedup.

    _job_id='rescore:{org}:{parcel}' — concurrent enqueues for the same
    (org, parcel) collapse to one execution per ARQ job_id-uniqueness guarantee.

    Returns:
        job_id str on enqueue, None on dedup (job already queued/running).
    """
    job_id = f"rescore:{organization_id}:{parcel_id}"
    job = await arq_pool.enqueue_job(
        "rescore_parcel",
        organization_id=str(organization_id),
        parcel_id=parcel_id,
        _job_id=job_id,
    )
    return job_id if job is not None else None
```

    parcel_id: str,

    # When any of {hypotheek_risico, eigendoms_complexiteit, koopwaarde_trend} is
    # missing (brk_records=[] or factor-specific input None), exclude its
    # default_weight from the risico-category total (100) and rescale gewicht +
    # bijdrage of non-missing risico factors proportionally so they sum back to 100.
    # Missing BRK factors stay in breakdown with missing=True (frontend cards can
    # show 'BRK-data nodig'); their bijdrage stays 0 — only non-missing risico
    # factors receive the rescale multiplier.
    missing_brk_keys = {
        k for k in _SPARSE_REDISTRIBUTABLE_RISICO
        if k in breakdown and breakdown[k].missing
    }
    if missing_brk_keys:
        missing_weight = sum(
            FACTOR_REGISTRY[k].default_weight for k in missing_brk_keys
        )
        active_weight = 100 - missing_weight
        if active_weight > 0:
            scale = 100.0 / active_weight
            for k, factor in breakdown.items():
                if (
                    factor.categorie == "risico"
                    and k not in missing_brk_keys
                    and not factor.missing

                    factor.gewicht = round(factor.gewicht * scale, 4)
                    factor.bijdrage = round(

    # ---------------------------------------------------------------------------
    # M050/S03 — per-parcel rescore job (D7 hook target)
    # ---------------------------------------------------------------------------


    async def rescore_parcel(
        ctx: dict[str, Any],
        *,
        organization_id: str,
        parcel_id: str,
    ) -> dict[str, Any]:
        """ARQ job — recompute Score v3 for one (org, parcel) pair.

        Triggered after BRK persist (D7) via enqueue_rescore_parcel. Idempotent
        via ARQ _job_id='rescore:{org}:{parcel}' — concurrent enqueues collapse.

        Args:
            ctx: ARQ job context (unused — required by ARQ signature).
            organization_id: caller org UUID stringified (ARQ JSON serialization).
            parcel_id: parcel identifier (e.g. NL.IMKAD.xxx).

        Returns:
            dict with parcel_id, updated, breakdown_keys, m050_s03_factors_at,
            duration_ms — surfaced in ARQ result-store for observability.
        """
        from app.models.scan_result import ScanResult
        from app.services.score_v3 import compute_score_v3
        from app.services.scoring_pipeline import enrich_parcel_data_with_brk

        org_uuid = uuid.UUID(organization_id)
        engine = create_async_engine(settings.DATABASE_URL)
        async_session_factory = async_sessionmaker(engine, expire_on_commit=False)

        started_at = time.time()
        try:
            async with async_session_factory() as db:
                stmt = (
                    select(ScanResult)
                    .where(ScanResult.parcel_id == parcel_id)
                    .order_by(ScanResult.created_at.desc())
                    .limit(1)
                )
                result = await db.execute(stmt)
                scan = result.scalar_one_or_none()
                if scan is None:
                    logger.warning(
                        "rescore_parcel: no scan_result for parcel=%s org=%s",
                        parcel_id, organization_id,
                    )
                    return {
                        "parcel_id": parcel_id,
                        "updated": False,
                        "reason": "no_scan_result",
                    }

                flags = dict(scan.flags or {})
                flags["parcel_id"] = parcel_id
                flags = await enrich_parcel_data_with_brk(flags, org_uuid, db)
                score_result = compute_score_v3(flags)
                flags["score_breakdown"] = {
                    k: {
                        "genormaliseerde_score": v.genormaliseerde_score,
                        "gewicht": v.gewicht,
                        "bijdrage": v.bijdrage,
                        "categorie": v.categorie,
                        "toelichting_kort": v.toelichting_kort,
                        "missing": v.missing,
                    }
                    for k, v in score_result.score_breakdown.items()
                }
                flags["m050_s03_factors_at"] = datetime.utcnow().isoformat()
                scan.flags = flags
                await db.commit()
        finally:
            await engine.dispose()

        return {
            "parcel_id": parcel_id,
            "updated": True,
            "breakdown_keys": list(score_result.score_breakdown.keys()),
            "m050_s03_factors_at": flags["m050_s03_factors_at"],
            "duration_ms": int((time.time() - started_at) * 1000),
        }


    async def enqueue_rescore_parcel(
        arq_pool: Any,
        organization_id: uuid.UUID,
        parcel_id: str,
    ) -> str | None:
        """Enqueue rescore_parcel via ARQ with deterministic _job_id for F7 dedup.

        Args:
            arq_pool: arq.connections.ArqRedis pool.
            organization_id: caller org UUID.
            parcel_id: parcel identifier.

        Returns:
            job_id str on enqueue, None on dedup (job already queued/running per
            ARQ _job_id-uniqueness guarantee).
        """
        job_id = f"rescore:{organization_id}:{parcel_id}"
        job = await arq_pool.enqueue_job(
            "rescore_parcel",
            organization_id=str(organization_id),
            parcel_id=parcel_id,
            _job_id=job_id,
        )
        return job_id if job is not None else None
    ```
