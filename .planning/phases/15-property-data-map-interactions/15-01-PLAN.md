    **1. Create `frontend/src/hooks/useAddressSuggest.ts`:**

    Debounced address suggest hook using TanStack Query:

    ```typescript
    export function useAddressSuggest(query: string) {
      // 1. Internal state for debounced query
      const [debouncedQuery, setDebouncedQuery] = useState("");

      // 2. Debounce effect: 300ms delay before updating debouncedQuery
      useEffect(() => {
        const timer = setTimeout(() => setDebouncedQuery(query), 300);
        return () => clearTimeout(timer);
      }, [query]);

      // 3. Get access token from useAuth()
      const { accessToken } = useAuth();

      // 4. TanStack Query with:
      //    - queryKey: ["address-suggest", debouncedQuery]
      //    - queryFn: calls suggestAddresses(debouncedQuery, accessToken, signal)
      //    - enabled: debouncedQuery.length >= 3
      //    - staleTime: 30_000 (30s — suggests don't change often)
      //    - signal from queryFn context for automatic cancellation
      return useQuery({
        queryKey: ["address-suggest", debouncedQuery],
        queryFn: async ({ signal }) => {
          return suggestAddresses(debouncedQuery, accessToken, signal);
        },
        enabled: debouncedQuery.length >= 3,
        staleTime: 30_000,
      });
    }
    ```

    Return type is `UseQueryResult<SuggestResult[]>`.

    **2. Create `frontend/src/hooks/usePropertyLookup.ts`:**

    Property lookup mutation hook using TanStack Query:

    ```typescript
    export function usePropertyLookup() {
      const { accessToken } = useAuth();
      const queryClient = useQueryClient();

      return useMutation({
        mutationFn: async (input: AddressInput) => {
          return lookupProperty(input, accessToken);
        },
        onSuccess: (data) => {
          // Cache the report by address ID for future lookups
          queryClient.setQueryData(
            ["property", data.address.nummeraanduiding_id],
            data
          );
        },
      });
    }
    ```

    Return type is `UseMutationResult<PropertyReportResponse, Error, AddressInput>`.

    Import useAuth from `@/lib/auth/useAuth`. Import API functions from `@/lib/api`. Import types from `@/lib/types/property`.
