**2. Create `frontend/src/hooks/useEigendom.ts`:**

Ephemeral React Query hook per research pattern:

```typescript
import { useQuery } from '@tanstack/react-query'
import { apiFetch } from '@/lib/api'
import type { BRKOwnershipResponse } from '@/lib/types/brk'

export function useEigendomLookup(parcelId: string) {
  return useQuery({
    queryKey: ['brk-eigendom', parcelId],
    queryFn: async () => {
      const res = await apiFetch(`/api/scans/parcels/${parcelId}/eigendom`)
      if (!res.ok) throw new Error('Eigendom ophalen mislukt')
      return res.json() as Promise<BRKOwnershipResponse>
    },
    enabled: false,    // Manual trigger only via refetch()
    staleTime: 0,      // Always stale
    gcTime: 0,         // Discard on unmount (Kadaster Tier 3)
    retry: false,      // Don't auto-retry paid API calls
  })
}
```

CRITICAL: `gcTime: 0` ensures BRK data is garbage collected immediately when modal unmounts. This is a Kadaster Tier 3 compliance requirement. Do NOT change these query options.
