```typescript
// Types
export interface MonthBucket {
    month: string        // "YYYY-MM"
    total: number
    toegewezen: number
    afgewezen: number
    gedeeltelijk: number
    klimaat_score: number
    avg_doorlooptijd: number | null
}

export interface TrendSummary {
    total_rulings: number
    avg_doorlooptijd: number | null
    succes_pct: number
    months_with_data: number
}

export interface TrendResponse {
    gemeente_naam: string
    months: MonthBucket[]
    summary: TrendSummary
}
```

Add `fetchTrendData` function following the existing `getGemeenteRulingsFeed` pattern:
- `GET ${API_BASE}/jurisprudence/trend/${encodeURIComponent(gemeenteNaam)}`
- Bearer token auth
- Returns `TrendResponse`

2. In `frontend/src/hooks/useJurisprudence.ts`, add:

```typescript
export function useTrendData(gemeenteNaam: string) {
    const { accessToken } = useAuth()
    return useQuery<TrendResponse, Error>({
        queryKey: ['jurisprudence', 'trend', gemeenteNaam],
        queryFn: () => fetchTrendData(gemeenteNaam, accessToken!),
        enabled: !!accessToken && gemeenteNaam.length > 0,
        staleTime: 10 * 60_000, // 10 min -- trend data changes slowly
    })
}
```

Import `fetchTrendData` and `TrendResponse` from `@/lib/api`. Export the hook.
