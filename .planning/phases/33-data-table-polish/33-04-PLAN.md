
  4. Create `handleRowHover` callback:
     ```typescript
     const handleRowHover = useCallback((parcelId: string | null) => {
       if (!mapRef.current) return
       const map = mapRef.current

       // Clear previous hover
       if (hoveredParcelRef.current !== null) {
         map.setFeatureState(
           { source: 'parcel-scores', sourceLayer: 'parcel_scores', id: hoveredParcelRef.current },
           { hover: false }
         )
         hoveredParcelRef.current = null
       }

       if (parcelId) {
         // Find the feature by parcel_id property to get its numeric MVT ID
         const features = map.querySourceFeatures('parcel-scores', {
           sourceLayer: 'parcel_scores',
           filter: ['==', ['get', 'parcel_id'], parcelId],
         })
         if (features.length > 0 && features[0].id !== undefined) {
           map.setFeatureState(
             { source: 'parcel-scores', sourceLayer: 'parcel_scores', id: features[0].id },
             { hover: true }
           )
           hoveredParcelRef.current = features[0].id
         }
       }
     }, [mapRef])
     ```
     CRITICAL: Use `features[0].id` (numeric MVT ID), NOT `parcelId` (string UUID). Pitfall 3 from research.

  5. Pass `onRowHover={handleRowHover}` to the `ResultsTable` in ResultsLayout:
