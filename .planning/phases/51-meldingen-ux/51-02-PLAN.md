**Frontend: Create useInfiniteAllChanges hook**

In `frontend/src/hooks/useNotifications.ts`, add a new hook `useInfiniteAllChanges` using TanStack Query's `useInfiniteQuery`:

```typescript
import { useInfiniteQuery } from '@tanstack/react-query'

const PAGE_SIZE = 25

export function useInfiniteAllChanges(
  includeMatches = true,
  gemeenteNaam?: string
) {
  const { accessToken } = useAuth()

  return useInfiniteQuery<ChangeRecordListResponse, Error>({
    queryKey: ['changes', 'all-infinite', includeMatches, gemeenteNaam],
    queryFn: ({ pageParam = 0 }) =>
      fetchAllChanges(accessToken, PAGE_SIZE, pageParam as number, includeMatches, gemeenteNaam),
    initialPageParam: 0,
    getNextPageParam: (lastPage, _allPages, lastPageParam) => {
      if (lastPage.has_more) {
        return (lastPageParam as number) + PAGE_SIZE
      }
      return undefined // No more pages
    },
    enabled: !!accessToken,
    staleTime: 60_000,
  })
}
```

Keep the existing `useAllChanges` hook unchanged — `ChangeTimeline.tsx` and other consumers may still use it. The meldingen page will switch to `useInfiniteAllChanges`.

**read_change_ids across pages**: Each page response includes its own `read_change_ids`. The frontend must merge them across all pages. This will be handled in Task 2 when wiring the meldingen page.

**Add "Load more" trigger at the bottom of the feed**

Add an intersection observer-based scroll sentinel after the last date group:

```typescript
import { useRef, useEffect } from 'react'

// Inside the component:
const sentinelRef = useRef<HTMLDivElement>(null)
const { fetchNextPage, hasNextPage, isFetchingNextPage } = // from useInfiniteAllChanges

useEffect(() => {
  if (!sentinelRef.current || !hasNextPage) return
  const observer = new IntersectionObserver(
    ([entry]) => {
      if (entry?.isIntersecting && hasNextPage && !isFetchingNextPage) {
        fetchNextPage()
      }
    },
    { rootMargin: '200px' } // Trigger 200px before reaching the bottom
  )
  observer.observe(sentinelRef.current)
  return () => observer.disconnect()
}, [hasNextPage, isFetchingNextPage, fetchNextPage])

// In JSX, after the date-grouped feed:
{hasNextPage && (
  <div ref={sentinelRef} className="flex items-center justify-center py-4">
    {isFetchingNextPage ? (
      <div className="flex items-center gap-2 text-sm text-grey-400">
        <Skeleton className="h-4 w-4 rounded-full" />
        Meer laden...
      </div>
    ) : (
      <span className="text-xs text-grey-600">Scroll voor meer</span>
    )}
  </div>
)}
```
