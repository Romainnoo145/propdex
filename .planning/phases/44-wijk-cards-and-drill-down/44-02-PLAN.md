**3. Create WijkCardWithNav sub-component** (below the main `ProjectList` component, same pattern as existing `WijkScanList` sub-component):

```typescript
function WijkCardWithNav({
  wijk,
  svgData,
  unreadCount,
  navigate,
}: {
  wijk: ProjectResponse
  svgData: { d: string; viewBox: string } | null
  unreadCount: number
  navigate: ReturnType<typeof useNavigate>
}) {
  const { data } = useProjectScans(wijk.id)

  const handleClick = () => {
    const latestScan = data?.scans.find((s) => s.status === 'completed')
    if (!latestScan) {
      toast.info('Nog geen scan beschikbaar voor deze wijk')
      return
    }
    navigate({ to: '/resultaten', search: { scan_id: latestScan.scan_id } })
  }

  return (
    <WijkCard
      wijk={wijk}
      svgData={svgData}
      unreadCount={unreadCount}
      onClick={handleClick}
    />
  )
}
```

Note: `toast` is already imported in ProjectList.tsx. `WijkCard` is imported in step 2. `useProjectScans` is already imported. This sub-component is its own React component, so calling `useProjectScans` at the top level is valid (not inside a .map() callback).
