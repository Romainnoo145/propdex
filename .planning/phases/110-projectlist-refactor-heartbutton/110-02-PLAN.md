**2. Create `HeartButton.tsx`:**

```tsx
import { Heart } from 'lucide-react'
import { useMemo } from 'react'
import { useParcelFavorites, useToggleFavorite } from '@/hooks/useFavorites'
import type { ScanResultItem } from '@/lib/types/scan'

interface HeartButtonProps {
  parcel: ScanResultItem
}

export function HeartButton({ parcel }: HeartButtonProps) {
  const favoritesQuery = useParcelFavorites()
  const toggleFavorite = useToggleFavorite()

  const isFavorited = useMemo(
    () => (favoritesQuery.data ?? []).some((f) => f.parcel_id === parcel.parcel_code),
    [favoritesQuery.data, parcel.parcel_code]
  )

  if (!parcel.parcel_code) return null  // Guard: no Kadaster ID = can't favorite

  return (
    <button
      type="button"
      onClick={() => toggleFavorite.mutate({ parcel, isFavorited })}
      disabled={toggleFavorite.isPending}
      className="flex h-8 w-8 items-center justify-center rounded-full bg-black/30 text-white backdrop-blur-sm transition-colors hover:bg-black/50"
      aria-label={isFavorited ? 'Verwijder uit portfolio' : 'Toevoegen aan portfolio'}
    >
      <Heart
        className={`h-4 w-4 transition-colors ${isFavorited ? 'fill-amber-400 text-amber-400' : ''}`}
      />
    </button>
  )
}
```

Use `fill-amber-400 text-amber-400` for the filled state (matches Warm Amber from Aura palette). If `fill-aura-amber` is defined as a Tailwind custom color in the project, use that instead — check `tailwind.config` for aura color definitions.
