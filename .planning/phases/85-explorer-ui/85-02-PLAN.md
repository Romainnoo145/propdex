    ```typescript
    interface ExplorerRuling {
      ecli: string
      court_code: string

      summary: string | null

    interface FacetItem {
      value: string
      count: number
    }

    interface ExplorerResponse {
      results: ExplorerRuling[]
      total_count: number
      has_more: boolean
      facets: {
        gemeente: FacetItem[]
        vergunningstype: FacetItem[]
        uitkomst: FacetItem[]
        rechtsgrond: FacetItem[]
        rechtbank_niveau: FacetItem[]
      }
    }

    interface ExplorerFilters {

      uitkomst?: string
      rechtsgrond?: string
      rechtbank_niveau?: string
      periode?: string
    }
    ```

    2. **useExplorer.ts** — Create hook using `useInfiniteQuery` pattern from `useNotifications.ts`:

    ```typescript
    const PAGE_SIZE = 20

    export function useExplorerResults(filters: ExplorerFilters) {
      const { accessToken } = useAuth()

      return useInfiniteQuery<ExplorerResponse, Error>({
        queryKey: ['explorer', filters],
        queryFn: ({ pageParam = 0 }) =>
          fetchExplorerResults(accessToken!, filters, PAGE_SIZE, pageParam as number),
        initialPageParam: 0,
        getNextPageParam: (lastPage, _allPages, lastPageParam) => {
          if (lastPage.has_more) {
            return (lastPageParam as number) + PAGE_SIZE
          }
          return undefined
        },
        enabled: !!accessToken,
        staleTime: 5 * 60_000,
      })
    }
    ```

    The hook returns all pages flattened + facets from the latest page (facets update with each fetch since filters may change). Export the ExplorerFilters and ExplorerRuling types.
