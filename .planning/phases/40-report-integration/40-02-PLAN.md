```typescript
/**
 * Generate a full property intelligence report from a scan result parcel
 *
 * POST /api/reports/from-parcel
 *
 * Triggers the property aggregator pipeline (BAG, PDOK, DSO, RCE, SRU)
 * for the parcel's address and saves the result as a report.
 *
 * @param parcelId - ParcelCache UUID
 * @param accessToken - JWT access token from useAuth()
 * @param options - Optional notes and project_id
 * @returns Created report with full details
 */
export async function generateReportFromParcel(
  parcelId: string,
  accessToken: string | null,
  options?: { notes?: string; project_id?: string }
): Promise<ReportDetailResponse> {
  return apiFetch<ReportDetailResponse>(`${REPORTS_BASE}/from-parcel`, {
    method: 'POST',
    accessToken,
    body: JSON.stringify({
      parcel_id: parcelId,
      ...(options?.notes && { notes: options.notes }),
      ...(options?.project_id && { project_id: options.project_id }),
    }),
  })
}
```

**2. Add mutation hook to `frontend/src/hooks/useReports.ts`:**

Add the `generateReportFromParcel` import to the existing import block from `@/lib/api`.

Add this hook after `useExportPDF` (around line 159):

```typescript
/**
 * Generate a property report from a scan result parcel
 *
 * Calls POST /api/reports/from-parcel.
 * Invalidates report list cache on success so the new report
 * appears immediately in the Rapporten view.
 */
export function useGenerateReportFromParcel() {
  const { accessToken } = useAuth()
  const queryClient = useQueryClient()

  return useMutation<ReportDetailResponse, Error, string>({
    mutationFn: (parcelId: string) =>
      generateReportFromParcel(parcelId, accessToken),
    onSuccess: () => {
      queryClient.invalidateQueries({ queryKey: ['reports'] })
    },
  })
}
```

The mutation accepts a `string` (parcel UUID) and returns `ReportDetailResponse`. On success it invalidates `['reports']` so the Rapporten view picks up the new report.
