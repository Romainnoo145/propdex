```typescript
import { useState, useCallback, useEffect } from 'react'
import type { PersistedMapLayerState, BaseMapLayerId, ParcelOverlayLayerId } from './layers'
import { readPersistedMapLayerState, writePersistedMapLayerState } from './layers'

export interface MapLayerState extends PersistedMapLayerState {
  // PersistedMapLayerState already has:
  // basemap, showParcels, showGemeenten, showBestemmingsplan, showBAGPanden, showScannedScope, activeParcelOverlay
}

export interface MapLayerActions {
  setBasemap: (v: BaseMapLayerId) => void
  toggleParcels: () => void
  setParcels: (v: boolean) => void
  toggleGemeenten: () => void
  setGemeenten: (v: boolean) => void
  toggleBestemmingsplan: () => void
  setBestemmingsplan: (v: boolean) => void
  setActiveParcelOverlay: (v: ParcelOverlayLayerId | null) => void
}

export function useMapLayerState() {
  const [state, setState] = useState<MapLayerState>(readPersistedMapLayerState)

  useEffect(() => {
    writePersistedMapLayerState(state)
  }, [state])

  const setBasemap = useCallback((v: BaseMapLayerId) =>
    setState(s => ({ ...s, basemap: v })), [])

  const toggleParcels = useCallback(() =>
    setState(s => ({ ...s, showParcels: !s.showParcels })), [])

  const setParcels = useCallback((v: boolean) =>
    setState(s => ({ ...s, showParcels: v })), [])

  const toggleGemeenten = useCallback(() =>
    setState(s => ({ ...s, showGemeenten: !s.showGemeenten })), [])

  const setGemeenten = useCallback((v: boolean) =>
    setState(s => ({ ...s, showGemeenten: v })), [])

  const toggleBestemmingsplan = useCallback(() =>
    setState(s => ({ ...s, showBestemmingsplan: !s.showBestemmingsplan })), [])

  const setBestemmingsplan = useCallback((v: boolean) =>
    setState(s => ({ ...s, showBestemmingsplan: v })), [])

  const setActiveParcelOverlay = useCallback((v: ParcelOverlayLayerId | null) =>
    setState(s => ({ ...s, activeParcelOverlay: v })), [])

  const actions: MapLayerActions = {
    setBasemap, toggleParcels, setParcels,
    toggleGemeenten, setGemeenten,
    toggleBestemmingsplan, setBestemmingsplan,
    setActiveParcelOverlay,
  }

  return { state, actions } as const
}
```

Key design decisions:
- `toggle*` for user-facing button clicks (no args), `set*` for programmatic control (explicit value). Both patterns needed.
- Actions object returned as stable ref (useCallback on all setters).
- `MapLayerState` extends `PersistedMapLayerState` for forward compatibility (can add non-persisted state like `mapReady` later).
- No `showBAGPanden` or `showScannedScope` setters yet — these are not toggled by the user in the current UI. Add only when needed.

interface LayerToggleProps {
  // State (from useMapLayerState)
  basemap: BaseMapLayerId
  showParcels: boolean
  showGemeenten: boolean
  showBestemmingsplan: boolean

  // Handlers (from useMapLayerState.actions or map.tsx callbacks)
  onBasemapChange: (v: BaseMapLayerId) => void
  onParcelsToggle: () => void
  onGemeentenToggle: () => void
  onBestemmingsplanToggle: () => void

  // Map readiness (needed for one-time layer init)

  // Auto-enable parcels trigger (will be removed in Plan 02, keep for now)

}

**What to REMOVE from LayerToggle:**
1. All `useState` calls for `showParcels`, `showGemeenten`, `showBestemmingsplan`, `showBAGPanden`, `showScannedScope`, `showScore`, `showBuiltStatus`, `showReportStatus`, `showVacancy`, `showCapacity`, `activeLayer`.
2. All dead callback props: `onBAGPandenToggle`, `onScoreToggle`, `onBuiltStatusToggle`, `onReportStatusToggle`, `onVacancyToggle`, `onCapacityToggle`, `onScannedScopeToggle`.
3. All dead boolean props: `showScoreToggle`, `showBuiltStatusToggle`, `showReportStatusToggle`, `showVacancyToggle`, `showCapacityToggle`, `showBAGPandenToggle`, `showScannedScopeToggle`.
4. The `writePersistedMapLayerState` call inside the useEffect — persistence is now in the hook.
5. The `hasScoreData` prop — dead code.

**What to KEEP:**
1. The toggle bar rendering (Kaart | Satellite || Gemeenten || Percelen (Bestemmingsplan)).
2. The `isStyleReady` local state guard — this is a transient map readiness concern, not toggle state.
3. The `didInitSyncRef` one-time init effect — but simplify it. After removing internal state, it only needs to call the initial map layer visibility setters once when the map is ready. It should read the props (not internal state) to set initial visibility.
4. The conditional rendering of Bestemmingsplan only when `showParcels` is true (already in uncommitted changes).
5. The `autoEnableParcelsKey` effect for now — Plan 02 will move this to map.tsx.

**Basemap handling:** `activeLayer` (useState for basemap) becomes `basemap` prop. The `handleLayerSwitch` function calls `onBasemapChange(layerId)` instead of `setActiveLayer(layerId)`.
