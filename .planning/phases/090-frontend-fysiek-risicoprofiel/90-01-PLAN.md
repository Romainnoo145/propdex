    ```typescript
    export async function fetchParcelRiskProfile(parcelId: string): Promise<ParcelRiskProfileResponse> {
      const res = await fetch(`${SCANS_BASE}/parcels/${parcelId}/risk-profile`, {
        credentials: 'include',
      })
      if (!res.ok) throw new Error('Failed to fetch risk profile')
      return res.json()
    }
    ```

    ```typescript
    export function useParcelRiskProfile(parcelId: string | null, options?: { enabled?: boolean }) {
      return useQuery({

        queryFn: () => fetchParcelRiskProfile(parcelId!),
        enabled: (options?.enabled ?? true) && !!parcelId,
        staleTime: 5 * 60 * 1000, // 5 min — backend caches underlying data
      })
    }
    ```
