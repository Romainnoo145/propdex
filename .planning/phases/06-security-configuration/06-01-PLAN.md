```python
"""Security audit logging utilities for multi-tenant access control."""

import logging
import uuid
from datetime import datetime


# Dedicated audit logger — can be configured independently from application logs
audit_logger = logging.getLogger("audit.security")


async def log_access_denial(
    user_id: uuid.UUID | None,
    organization_id: uuid.UUID,
    resource_type: str,
    resource_id: uuid.UUID | str,
    reason: str,
) -> None:
    """
    Log a security-relevant access denial event.

    Called when a user attempts to access a resource that either:
    - Does not exist (legitimate 404)
    - Belongs to another organization (cross-tenant attempt, also returns 404)

    These logs enable security monitoring and compliance auditing.
    The application returns 404 for both cases to prevent resource enumeration.

    Args:
        user_id: Authenticated user UUID (None if unauthenticated)
        organization_id: User's organization UUID
        resource_type: Type of resource (e.g., "saved_report", "project")
        resource_id: Resource UUID or identifier
        reason: Description of denial (e.g., "not_found_or_wrong_org")
    """
    audit_logger.warning(
        "access_denied: user=%s org=%s resource=%s/%s reason=%s",
        str(user_id) if user_id else "anonymous",
        str(organization_id),
        resource_type,
        str(resource_id),
        reason,
        extra={
            "event_type": "access_denied",
            "timestamp": datetime.utcnow().isoformat(),
            "user_id": str(user_id) if user_id else None,
            "organization_id": str(organization_id),
            "resource_type": resource_type,
            "resource_id": str(resource_id),
            "reason": reason,
        },
    )


async def log_rls_context_set(
    organization_id: uuid.UUID,
) -> None:
    """
    Log RLS context being set for a database session (debug level).

    Useful for troubleshooting tenant isolation issues.
    """
    audit_logger.debug(
        "rls_context_set: org=%s",
        str(organization_id),
        extra={
            "event_type": "rls_context_set",
            "organization_id": str(organization_id),
        },
    )
```

IMPORTANT: Use `datetime.utcnow()` per project convention (MEMORY.md), NOT `datetime.now(timezone.utc)`.
