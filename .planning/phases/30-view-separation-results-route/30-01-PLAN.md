**ResultsShell.tsx — Add `fullWidth` prop:**

Add an optional `fullWidth?: boolean` prop to `ResultsShellProps`. When `fullWidth` is true, skip the inner `max-w-5xl mx-auto px-6 py-8` container and render children directly inside the outer div. When false/undefined, keep the existing inner container behavior (backward compatible).

Implementation:
```tsx
interface ResultsShellProps {
  children: React.ReactNode
  className?: string
  fullWidth?: boolean  // NEW: bypass inner max-w-5xl container
}

export default function ResultsShell({ children, className, fullWidth }: ResultsShellProps) {
  return (
    <div className={cn('w-full h-full overflow-y-auto bg-grey-950', className)}>
      {fullWidth ? children : (
        <div className="max-w-5xl mx-auto px-6 py-8">
          {children}
        </div>
      )}
    </div>
  )
}
```
