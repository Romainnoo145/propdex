```python
# Hash (org_id, parcel_id) → signed bigint for advisory-lock key.
# Python hash() is process-local — use a stable hash (hashlib + int conversion).
lock_key = int.from_bytes(
    hashlib.sha256(f"{organization_id}:{parcel_id}".encode()).digest()[:8],
    byteorder="big",
```

    **Function 1 — `_acquire_advisory_lock`** (D5):

    ```python
    async def _acquire_advisory_lock(
        db: AsyncSession,
        organization_id: uuid.UUID,
        parcel_id: str,
    ) -> bool:
        """D5: try-acquire pg_try_advisory_xact_lock(hash(org, parcel)).

        Returns True if acquired (caller is the winner). Returns False if contended
        (caller should wait via blocking variant, then re-read cache).

        Lock is transaction-scoped (_xact_): released automatically at commit/rollback.
        """
        lock_key = int.from_bytes(
            hashlib.sha256(f"{organization_id}:{parcel_id}".encode()).digest()[:8],
            byteorder="big",
            signed=True,
        )
        # SET LOCAL + asyncpg cannot use parameterized queries → f-string with int (safe).
        result = await db.execute(
            sa.text(f"SELECT pg_try_advisory_xact_lock({lock_key}) AS acquired")
        )
        return bool(result.scalar_one())


    async def _wait_for_advisory_lock(
        db: AsyncSession,
        organization_id: uuid.UUID,
        parcel_id: str,
    ) -> None:
        """D5 contended-path: block until lock free (winner has committed)."""
        lock_key = int.from_bytes(
            hashlib.sha256(f"{organization_id}:{parcel_id}".encode()).digest()[:8],
            byteorder="big",
            signed=True,
        )
        await db.execute(sa.text(f"SELECT pg_advisory_xact_lock({lock_key})"))
    ```

    **Function 2 — `get_latest_for_parcel`** (D3 read-path):

    ```python
    async def get_latest_for_parcel(
        db: AsyncSession,
        organization_id: uuid.UUID,
        parcel_id: str,
    ) -> list[BRKOwnershipRecordRead]:
        """Return all non-soft-deleted rows for (org, parcel), newest fetched_at first.

        Empty list = cache MISS (no rows ever, or all soft-deleted).
        Non-empty list with newest.fetched_at within CACHE_TTL_DAYS = cache HIT.
        Non-empty list older than CACHE_TTL_DAYS = stale → refetch.

        Backed by partial composite index idx_brk_latest (D15, Plan 02-01).
        """
        result = await db.execute(
            select(BrkOwnershipRecord)
            .where(
                BrkOwnershipRecord.organization_id == organization_id,
                BrkOwnershipRecord.parcel_id == parcel_id,
                BrkOwnershipRecord.soft_deleted_at.is_(None),
            )
            .order_by(BrkOwnershipRecord.fetched_at.desc())
        )
        rows = result.scalars().all()
        return [BRKOwnershipRecordRead.model_validate(r) for r in rows]
    ```

    **Function 3 — `_is_cache_fresh`** helper:

    ```python
    def _is_cache_fresh(records: list[BRKOwnershipRecordRead]) -> bool:
        """True if newest fetched_at is within CACHE_TTL_DAYS (D3)."""
        if not records:
            return False
        newest = max(r.fetched_at for r in records)
        return (datetime.utcnow() - newest) < timedelta(days=CACHE_TTL_DAYS)
    ```

    **Function 5 — `fetch_and_persist`** (main orchestration, D4 + D5 + D6 + D8):

    ```python
    async def fetch_and_persist(
        db: AsyncSession,
        organization_id: uuid.UUID,
        parcel_id: str,
    ) -> list[BRKOwnershipRecordRead]:
        """D4 two-txn boundary; D5 advisory-lock dedup; D6 factory; D8 quota hook.

        Flow:
          Txn1: try-acquire lock → cache-check → commit (releases lock).
            If contended: wait for lock → re-read cache → return cached (winner populated).
          HTTP-Kadaster (outside any txn): resolve creds → factory client → get_eigendom.
          Txn2: parse → persist_record per rechtspersoon/overheid → commit.
          AFTER Txn2 commit: quota_service.record_brk_lookup (cache-miss path only).

        D12: raises typed Python exceptions only (no fastapi imports here).
        D9: returns list[BRKOwnershipRecordRead] from get_latest_for_parcel after persist.
        """
        # === Txn 1 ===
        acquired = await _acquire_advisory_lock(db, organization_id, parcel_id)
        if not acquired:
            await _wait_for_advisory_lock(db, organization_id, parcel_id)
            cached = await get_latest_for_parcel(db, organization_id, parcel_id)
            await db.commit()  # release lock + close txn
            if _is_cache_fresh(cached):
                logger.info(
                    "brk.fetch.cache_hit_via_lock_wait org=%s parcel=%s",
                    organization_id, parcel_id,
                )
                return cached
            # cached not fresh — winner found no rechtspersoon-eigenaar. Return empty.
            return cached

        cached = await get_latest_for_parcel(db, organization_id, parcel_id)
        await db.commit()  # release lock + close Txn1

        if _is_cache_fresh(cached):
            logger.info(
                "brk.fetch.cache_hit org=%s parcel=%s rows=%d",
                organization_id, parcel_id, len(cached),
            )
            return cached  # D3: no HTTP, no quota spend

        # === HTTP-Kadaster (no DB locks held) ===
        # Open a NEW session-scoped DB read for credentials only.
        creds: BrkCredentials = await get_brk_credentials(db)
        client = build_brk_client(creds)

        # Mirror property_aggregator.fetch_brk_ownership's http_client construction:
    ```
