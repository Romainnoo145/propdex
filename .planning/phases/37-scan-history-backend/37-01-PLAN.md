class ScanListItem(BaseModel):
    """Schema for a single scan in the history list."""
    scan_id: uuid.UUID = Field(description="AreaScan UUID")

**service.py** -- Add `list_scans` function at the END of the file (after `list_scan_results`):

```python
async def list_scans(
    organization_id: uuid.UUID,
    offset: int,
    limit: int,
    session: AsyncSession,
    project_id: uuid.UUID | None = None,
    status_filter: str | None = None,
) -> tuple[list[tuple], int]:
    """
    List AreaScans with parcel_count from scan_results.

    Returns list of (AreaScan, parcel_count) tuples and total count.
    Uses COUNT(scan_results) for parcel_count -- more accurate than
    parcels_total which reflects discovered parcels before qualification.

    Args:
        organization_id: Tenant organization ID
        offset: Pagination offset
        limit: Page size
        session: Database session
        project_id: Optional filter by project
        status_filter: Optional filter by scan status

    Returns:
        tuple[list[tuple], int]: (AreaScan, parcel_count) rows and total count
    """
    parcel_count_subq = (
        select(func.count(ScanResult.id))
        .where(ScanResult.scan_id == AreaScan.id)
        .scalar_subquery()
    )

    query = (
        select(AreaScan, parcel_count_subq.label("parcel_count"))
        .where(AreaScan.organization_id == organization_id)
    )

    if project_id is not None:
        query = query.where(AreaScan.project_id == project_id)

    if status_filter is not None:
        query = query.where(AreaScan.status == status_filter)

    # Count total before pagination (derive from same query to share filters)
    count_query = select(func.count()).select_from(query.subquery())
    count_result = await session.execute(count_query)
    total = count_result.scalar() or 0

    # Apply ordering and pagination
    query = query.order_by(AreaScan.created_at.desc()).offset(offset).limit(limit)
    result = await session.execute(query)
    return list(result.all()), total
```

Key decisions:
- Use `COUNT(scan_results)` not `parcels_total` -- per research Pitfall 2, this shows actual qualifying results
- Derive count_query from the main query's subquery so filters are automatically shared (Pitfall 4)
- Accept optional `project_id` param so Plan 37-02 can reuse this same function
- Accept optional `status_filter` for future frontend filtering (research Open Question 1)

```python
@router.get("", response_model=ScanListResponse)
@limiter.limit("60/minute")
async def list_scans(
    request: Request,
    offset: int = 0,
    limit: int = 50,
    status_filter: str | None = Query(
        default=None,
        alias="status",
        description="Filter by scan status (pending, running, completed, failed, cancelled)",
    ),
    current_user: User = Depends(get_current_active_user),
    organization_id: uuid.UUID = Depends(get_current_organization_id),
    db: AsyncSession = Depends(get_db),
) -> ScanListResponse:
    """
    List paginated scan history with metadata.

    Returns all scans for the authenticated organization, ordered by most recent
    first. Each scan includes gemeente name, wijk name, date, status, and the
    count of qualifying parcel results.

    Requires JWT authentication. Rate limited to 60 requests per minute.

    Args:
        request: FastAPI request object
        offset: Pagination offset (default 0)
        limit: Page size (default 50, max 100)
        status_filter: Optional filter by scan status
        current_user: Authenticated user
        organization_id: Tenant ID
        db: Database session

    Returns:
        ScanListResponse: Paginated scan list with total count
    """
    limit = min(limit, 100)

    scans, total = await service.list_scans(
        organization_id=organization_id,
        offset=offset,
        limit=limit,
        session=db,
        status_filter=status_filter,
    )

    scan_items = [
        ScanListItem(
            scan_id=scan.id,
            gemeente_naam=scan.gemeente_naam,
            gemeente_code=scan.gemeente_code,
            wijk_naam=scan.wijk_naam,
            scan_date=scan.created_at,
            status=scan.status,
            parcel_count=parcel_count or 0,
            project_id=scan.project_id,
        )
        for scan, parcel_count in scans
    ]

    return ScanListResponse(
        scans=scan_items,
        total=total,
        offset=offset,
        limit=limit,
    )
```

Use `Query(alias="status")` for the status_filter parameter so the URL reads naturally as `?status=completed` rather than `?status_filter=completed`. This avoids shadowing Python's built-in `status` while keeping a clean API surface.
