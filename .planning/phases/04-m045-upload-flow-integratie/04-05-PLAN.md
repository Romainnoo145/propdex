```python
    # ── Opportunistic probe-on-load (INT-04) ──────────────────────────────────
    # Identify Type A entries that came back as status='missing'. For each, run
    # probe_type_a_cascade in parallel. If a PolicyDocument row exists for this
    # gemeente+source_key (M045/S05-era upload), the probe flips status to 'live'
    # and upserts the data_source_status row — no manual Refresh needed.
    #
    # Advisory lock guards concurrent GET requests: if two requests race, one
    # probes and one skips (returns current DB state). Both callers see correct
    # data after the commit because they re-fetch below.
    #
    # Performance: first GET may add ~200-800ms for communes with several missing
    # Type A entries. Subsequent GETs are fast (rows no longer missing → no probe).
    missing_type_a_keys = [
        row.source_key
        for row in cascade_uploads
        if row.status == "missing"
    ]

    if missing_type_a_keys:

        async def _probe_one(source_key: str) -> None:
            lock_key = compute_lock_key(gemeente_code, source_key)
            lock_acquired: bool = bool(
                await db.scalar(
                    select(
                        func.pg_try_advisory_xact_lock(type_coerce(lock_key, BigInteger()))
                    )
                )
            )
            if not lock_acquired:
                return  # Another request is mid-probe — skip, caller gets fresh data soon.
            entry = get_entry(source_key)
            if entry is None:
                return
            probe_result = await probe_type_a_cascade(source_key, gemeente_code, db)
            if probe_result["status"] != "missing":
                status_row = await _load_or_create_row(gemeente_code, source_key, db)
                status_row.status = probe_result["status"]
                status_row.error_message = probe_result["error_message"]
                status_row.last_checked = datetime.utcnow()

        await asyncio.gather(*[_probe_one(sk) for sk in missing_type_a_keys])

        # Commit all upserts in one shot — consistent with refresh_source commit pattern.
        await db.commit()

        # Re-fetch DB rows to build accurate response (probes may have flipped statuses).
        db_rows = (
            await db.execute(
                select(DataSourceStatus).where(DataSourceStatus.gemeente_code == gemeente_code)
            )
        ).scalars().all()
        by_key = {r.source_key: r for r in db_rows}

        cascade_uploads = []
        open_apis = []
        paid_integrations = []
        for entry in DATA_SOURCE_KEYS:
            db_row = by_key.get(entry.key)
            row = _row_from_db(db_row, entry) if db_row else _synthesize_missing_row(entry)
            if entry.source_type == "A":
                cascade_uploads.append(row)
            elif entry.source_type == "B":
                open_apis.append(row)
            else:
                paid_integrations.append(row)
    # ── End opportunistic probe-on-load ───────────────────────────────────────
```
