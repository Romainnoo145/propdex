```typescript
interface Props {
  entityType: string
  entityCode: string
  sourceKey: string
  latestVersionNumber: number
  onView: (versionId: string, entityType: string, entityCode: string) => void
  onDownload: (versionId: string, filename: string, entityType: string, entityCode: string) => void
