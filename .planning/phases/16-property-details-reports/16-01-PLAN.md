       ```typescript
       // ReportSaveRequest — POST /api/reports body
       interface ReportSaveRequest {
         report_data: Record<string, unknown>  // Full PropertyReportResponse dict
         notes?: string | null
         project_id?: string | null  // UUID string
       }

       // ReportListItem — lightweight item in list view
       interface ReportListItem {
         id: string
         report_name: string
         postcode: string
         city: string
         notes: string | null
         project_id: string | null
         completeness_score: number
         created_at: string  // ISO datetime
         updated_at: string  // ISO datetime
       }

       // ReportListResponse — GET /api/reports
       interface ReportListResponse {
         reports: ReportListItem[]
         total: number
       }

       // ReportDetailResponse — GET /api/reports/{id}, POST /api/reports
       interface ReportDetailResponse {
         id: string
         report_name: string
         report_data: Record<string, unknown>  // Full PropertyReportResponse as JSONB
         postcode: string
         city: string
         notes: string | null
         project_id: string | null
         created_at: string
         updated_at: string
       }

       // ReportFreshnessResponse — GET /api/reports/{id}/freshness
       interface ReportFreshnessResponse {
         report_id: string
         has_updates: boolean
         updated_sections: Array<Record<string, unknown>>
       }

       // ReportUpdateNotes — PATCH /api/reports/{id}/notes
       interface ReportUpdateNotes {
         notes: string | null
       }

       // ReportAssignProject — PATCH /api/reports/{id}/project
       interface ReportAssignProject {
         project_id: string | null
       }

       // Sort parameters
       type SortField = 'created' | 'updated' | 'address'
       type SortOrder = 'asc' | 'desc'

       // Filters for report list query
       interface ReportFilters {
         city?: string
         project_id?: string
         search?: string
         sort_by?: SortField
         sort_order?: SortOrder
         skip?: number
         limit?: number
       }
       ```

       Export ALL types. Use `export interface` / `export type` syntax.
