3. Add a useEffect that shows the overlay when both `wijkGeojson` AND monitored projects are available. Place it after the existing useEffect blocks (after the "Cleanup highlight on unmount" effect around line 259):
   ```typescript
   // Show monitored wijk overlay when wijk GeoJSON loads and monitored projects exist
   useEffect(() => {
     if (!mapReady || !mapRef.current || !wijkGeojson) return

     // Extract monitored wijk names from projects data
     const monitoredWijkNames: string[] = []
     if (monitoredProjects?.projects) {
       for (const project of monitoredProjects.projects) {
         // Children are wijk projects with wijk_naam
         if (project.children) {
           for (const child of project.children) {
             if (child.wijk_naam) {
               monitoredWijkNames.push(child.wijk_naam)
             }
           }
         }
         // Also check top-level wijk projects (flat response fallback)
         if (project.wijk_naam) {
           monitoredWijkNames.push(project.wijk_naam)
         }
       }
     }

     if (monitoredWijkNames.length === 0) return

     addMonitoredWijkOverlay(mapRef.current, wijkGeojson, monitoredWijkNames)

     return () => {
       if (mapRef.current) {
         removeMonitoredWijkOverlay(mapRef.current)
       }
     }
   }, [mapReady, wijkGeojson, monitoredProjects])
   ```

   Key design choices:
   - Extracts `wijk_naam` from both `children` array (hierarchical response) AND top-level projects (flat response fallback)
   - Depends on `wijkGeojson` which is set when a gemeente is clicked and its wijken are loaded
   - Cleanup removes the overlay on unmount or when dependencies change
   - Uses the same imperative pattern as other map layers (ScanAreaHighlightLayer, WijkBoundariesLayer)

   # Group children under their parent projects

   children_by_parent: dict[uuid.UUID, list[ProjectResponse]] = {}
   root_projects: list[ProjectResponse] = []

   for proj_resp in project_responses:
       if proj_resp.parent_id is not None:
           parent_uuid = uuid.UUID(proj_resp.parent_id)
           children_by_parent.setdefault(parent_uuid, []).append(proj_resp)

   for root in root_projects:
