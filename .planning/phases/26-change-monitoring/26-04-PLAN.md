    **2. Add API functions** to `frontend/src/lib/api.ts`:
    Add `const MONITORING_BASE = '/api/monitoring'` constant.
    Add import for monitoring types.

    ```typescript
    export async function fetchUnreadCounts(accessToken: string | null): Promise<UnreadDetailResponse> {
      return apiFetch<UnreadDetailResponse>(`${MONITORING_BASE}/unread`, { accessToken, method: 'GET' })
    }

    export async function fetchProjectChanges(projectId: string, accessToken: string | null, limit = 50, offset = 0): Promise<ChangeRecordListResponse> {
      return apiFetch<ChangeRecordListResponse>(
        `${MONITORING_BASE}/changes?project_id=${projectId}&limit=${limit}&offset=${offset}`,
        { accessToken, method: 'GET' }
      )
    }

    export async function markProjectViewed(projectId: string, accessToken: string | null): Promise<void> {
      await apiFetch(`${MONITORING_BASE}/viewed/${projectId}`, { accessToken, method: 'POST' })
    }
    ```

    **3. Create `frontend/src/hooks/useNotifications.ts`:**
    ```typescript
    import { useQuery, useMutation, useQueryClient } from '@tanstack/react-query'
    import { useAuth } from '@/lib/auth/useAuth'
    import { fetchUnreadCounts, fetchProjectChanges, markProjectViewed } from '@/lib/api'

    export function useUnreadCounts() {
      const { accessToken } = useAuth()
      return useQuery({
        queryKey: ['notifications', 'unread'],
        queryFn: () => fetchUnreadCounts(accessToken),
        refetchInterval: 60_000,  // Poll every 60 seconds
        staleTime: 30_000,
        enabled: !!accessToken,
      })
    }

    export function useProjectChanges(projectId: string | null) {
      const { accessToken } = useAuth()
      return useQuery({
        queryKey: ['changes', projectId],
        queryFn: () => fetchProjectChanges(projectId!, accessToken),
        enabled: !!accessToken && !!projectId,
        staleTime: 60_000,
      })
    }

    export function useMarkViewed() {
      const { accessToken } = useAuth()
      const queryClient = useQueryClient()
      return useMutation({
        mutationFn: (projectId: string) => markProjectViewed(projectId, accessToken),
        onSuccess: () => {
          queryClient.invalidateQueries({ queryKey: ['notifications', 'unread'] })
        },
      })
    }
    ```
