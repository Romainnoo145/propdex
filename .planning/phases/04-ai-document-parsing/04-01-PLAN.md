  class BestemmingsregelExtract(BaseModel):
      """Single zoning rule extracted from bestemmingsplan PDF by AI."""
      artikel: str = Field(description="Article reference (e.g., 'Artikel 3.1.2 Wonen')")
      bestemming: str = Field(description="Zoning designation (e.g., 'Wonen', 'Gemengd')")
      toegestane_gebruiken: list[str] = Field(default_factory=list, description="Permitted uses")
      verboden_gebruiken: list[str] = Field(default_factory=list, description="Forbidden uses")
      bouwregels: BuildingDimensionsAI | None = Field(default=None, description="Building dimension rules")
      parkeereis: str | None = Field(default=None, description="Parking requirement text")
      milieu_restricties: list[str] = Field(default_factory=list, description="Environmental restrictions")
      bijzondere_voorwaarden: list[str] = Field(default_factory=list, description="Special conditions, exceptions")
      confidence: Literal["HIGH", "MEDIUM", "LOW"] = Field(description="Confidence for this specific rule")
      disclaimer: str | None = Field(default=None, description="Dutch warning for LOW-confidence items only")

  class BuildingDimensionsAI(BaseModel):
      """Building dimension constraints extracted by AI from bestemmingsplan text."""
      max_bouwhoogte: float | None = Field(default=None, description="Maximum building height in meters")
      max_goothoogte: float | None = Field(default=None, description="Maximum gutter height in meters")
      max_bebouwingspercentage: float | None = Field(default=None, description="Maximum coverage percentage (0-100)")
      max_oppervlakte: float | None = Field(default=None, description="Maximum floor area in m2")
      max_inhoud: float | None = Field(default=None, description="Maximum volume in m3")
      overige: dict[str, str] = Field(default_factory=dict, description="Other dimension rules as key-value pairs")

  class BestemmingsplanParsed(BaseModel):
      """Complete AI-parsed bestemmingsplan output from Claude."""
      plan_naam: str = Field(description="Plan name as stated in document")
      plan_idn: str = Field(description="IMRO plan identifier (NL.IMRO.{gemeente}.{plan}-{version})")
      regels: list[BestemmingsregelExtract] = Field(description="All extracted zoning rules")
      overall_confidence: Literal["HIGH", "MEDIUM", "LOW"] = Field(description="Overall parsing confidence")
      disclaimers: list[str] = Field(default_factory=list, description="Dutch disclaimers for ambiguous items")
      samenvatting: str = Field(description="Brief Dutch summary of key findings for developers")

  class AIParsingSection(BaseModel):
      """AI-parsed data section for inclusion in property report."""
      parsed_plan: BestemmingsplanParsed | None = Field(default=None, description="Parsed bestemmingsplan data")
      job_id: str | None = Field(default=None, description="ARQ job ID if parsing is in progress")
      status: Literal["complete", "in_progress", "queued", "failed", "not_available"] = Field(
          description="Current parsing status"

      error_message: str | None = Field(default=None, description="Error description if failed")

          default="AI-geanalyseerd \u2014 Controleer bij notaris voor investeringsbeslissingen",
          description="Section-level Dutch disclaimer banner"
      )

          description="Current job status"
      )
      progress_pct: int = Field(default=0, description="Estimated progress percentage (0-100)")
      message: str | None = Field(default=None, description="Human-readable status message")
      data: BestemmingsplanParsed | None = Field(default=None, description="Parsed data when complete")
      error: str | None = Field(default=None, description="Error message when failed")

  **1. Create app/models/pdf_document.py:**
  SQLAlchemy model for permanent PDF binary storage. Follow existing patterns from cached_response.py and base.py.

  ```python
  class PDFDocument(Base, TenantMixin, TimestampMixin):
      """
      Bestemmingsplan PDF storage with metadata.

      PDFs stored permanently linked to plan_idn for reuse across addresses.
      Multiple addresses in the same bestemmingsplan area share the same PDF.
      """
      __tablename__ = "pdf_documents"

      id: Mapped[uuid.UUID] = mapped_column(
          primary_key=True,
          server_default=text("gen_random_uuid()"),
      )
      plan_idn: Mapped[str] = mapped_column(
          String(150),
          nullable=False,
          comment="IMRO plan identifier (NL.IMRO.{gemeente}.{plan}-{version})"
      )
      file_data: Mapped[bytes] = mapped_column(
          LargeBinary,
          nullable=False,
          comment="PDF file bytes (max 100MB)"
      )
      file_size: Mapped[int] = mapped_column(
          Integer,
          nullable=False,
          comment="File size in bytes"
      )
      source_url: Mapped[str | None] = mapped_column(
          String(500),
          nullable=True,
          comment="Original download URL"
      )
      is_user_uploaded: Mapped[bool] = mapped_column(
          default=False,
          nullable=False,
          comment="True if manually uploaded, False if auto-downloaded"
      )

      __table_args__ = (
          Index("ix_pdf_documents_org_plan", "organization_id", "plan_idn"),
      )
  ```

  Import: uuid, datetime, `from sqlalchemy import Index, Integer, LargeBinary, String, text`, `from sqlalchemy.orm import Mapped, mapped_column`, `from app.models.base import Base, TenantMixin, TimestampMixin`.

  **2. Create app/models/parsing_job.py:**
  SQLAlchemy model for job metadata to enable tenant-scoped job access and prevent cross-tenant leakage.

  ```python
  class ParsingJob(Base, TenantMixin, TimestampMixin):
      """
      AI parsing job metadata for tenant-scoped job tracking.

      Stores job_id from ARQ alongside plan_idn and organization_id.
      Enables tenant isolation: job status endpoint validates organization ownership.
      """
      __tablename__ = "parsing_jobs"

      id: Mapped[uuid.UUID] = mapped_column(
          primary_key=True,
          server_default=text("gen_random_uuid()"),
      )
      job_id: Mapped[str] = mapped_column(
          String(100),
          nullable=False,
          unique=True,
          index=True,
          comment="ARQ job identifier"
      )
      plan_idn: Mapped[str] = mapped_column(
          String(150),
          nullable=False,
          comment="IMRO plan identifier being parsed"
      )
      status: Mapped[str] = mapped_column(
          String(20),
          nullable=False,
          default="queued",
          comment="Job status: queued, in_progress, complete, failed"
      )
      error_message: Mapped[str | None] = mapped_column(
          String(1000),
          nullable=True,
          comment="Error message if job failed"
      )

      __table_args__ = (
          Index("ix_parsing_jobs_org_plan", "organization_id", "plan_idn"),
      )
  ```
