    ```tsx
    /**
     * StatusPill — M049/S03
     *
     * Renders a data-source status as an Aura-conform pill in one of 4 colours:
     *   live          → Vitality Green
     *   broken        → red (+ tooltip with error_message)
     *   missing       → Warm Amber
     *   not_applicable→ neutral slate
     *
     * Tokens locked in 03-CONTEXT.md §3 — do not swap for Tailwind defaults.
     */
    import {
      Tooltip,
      TooltipContent,
      TooltipProvider,
      TooltipTrigger,
    } from '@/components/ui/tooltip'
    import type { SourceStatus } from '@/lib/types/data-sources-admin'

    interface Props {
      status: SourceStatus
      errorMessage?: string | null
    }

    const LABELS: Record<SourceStatus, string> = {
      live: 'Live',
      broken: 'Foutmelding',
      missing: 'Ontbreekt',
      not_applicable: 'N.v.t.',
    }

    const CLASSES: Record<SourceStatus, string> = {
      live: 'bg-[#22C55E]/15 text-[#22C55E]',
      broken: 'bg-red-100 text-red-700',
      missing: 'bg-[#F9A826]/15 text-[#F9A826]',
      not_applicable: 'bg-slate-100 text-slate-500',
    }

    export function StatusPill({ status, errorMessage }: Props) {
      const pill = (
        <span
          className={`inline-flex items-center rounded-full px-2.5 py-0.5 text-[11px] font-semibold ${CLASSES[status]}`}
          data-status={status}
        >
          {LABELS[status]}
        </span>
      )

      if (status === 'broken' && errorMessage) {
        return (
          <TooltipProvider delayDuration={150}>
            <Tooltip>
              <TooltipTrigger asChild>{pill}</TooltipTrigger>
              <TooltipContent side="top" className="max-w-xs text-[11px]">
                {errorMessage}
              </TooltipContent>
            </Tooltip>
          </TooltipProvider>
        )
      }

      return pill
    }
    ```

    NOTE: if `@/components/ui/tooltip` exports different names (e.g., shadcn alternate naming), adjust the import to match. Run `grep "^export" frontend/src/components/ui/tooltip.tsx` to confirm.

    Step A — Create `frontend/src/components/settings/data-bronnen/DataSourceRow.tsx`:

    ```tsx
    /**
     * DataSourceRow — M049/S03
     *
     * Pure-Glass card row for one data-source slot. Composes:
     *   [label + source_type badge]  [StatusPill]  [last_checked relative]  [actions]
     *
     * Action handling:
     *   - refresh action wired to useRefreshDataSource mutation
     *   - other actions (upload / view_versions / configure_credential) are placeholders
     *     for S04/S05; rendered as disabled buttons with Radix Tooltip when not enabled
     *
     * Aura tokens locked in 03-CONTEXT.md §5.
     */
    import { History, Loader2, RefreshCw, Settings2, Upload } from 'lucide-react'
    import { Link } from '@tanstack/react-router'
    import {
      Tooltip,
      TooltipContent,
      TooltipProvider,
      TooltipTrigger,
    } from '@/components/ui/tooltip'
    import { formatRelativeNL } from '@/lib/utils/datetime'
    import { useRefreshDataSource } from '@/lib/queries/data-sources-admin'
    import type {
      Action,
      ActionType,
      DataSourceRow as DataSourceRowDTO,
      SourceType,
    } from '@/lib/types/data-sources-admin'
    import { StatusPill } from './StatusPill'

    interface Props {
      gemeenteCode: string
      row: DataSourceRowDTO
    }

    const SOURCE_TYPE_BADGE: Record<SourceType, { className: string; label: string }> = {
      A: { className: 'bg-[#D1E2D8]/40 text-[#0E1111]', label: 'Beleid' },
      B: { className: 'bg-slate-100 text-slate-700', label: 'API' },
      C: { className: 'bg-[#F9A826]/15 text-[#F9A826]', label: 'Integratie' },
    }

    const ACTION_ICON: Record<ActionType, typeof Upload> = {
      upload: Upload,
      refresh: RefreshCw,
      view_versions: History,
      configure_credential: Settings2,
    }

    export function DataSourceRow({ gemeenteCode, row }: Props) {
      const refreshMutation = useRefreshDataSource()
      const isRefreshing =
        refreshMutation.isPending &&
        refreshMutation.variables?.sourceKey === row.source_key &&
        refreshMutation.variables?.gemeenteCode === gemeenteCode

      const badge = SOURCE_TYPE_BADGE[row.source_type]

      return (
        <div
          className="bg-white/60 backdrop-blur-[24px] border border-white/20 rounded-2xl p-4 flex flex-col md:flex-row md:items-center gap-4"
          data-source-key={row.source_key}
        >
          {/* Column 1: label + type badge */}
          <div className="flex items-center gap-2 min-w-0">
            <span className="text-[13px] font-semibold text-[#0E1111] truncate">
              {row.label_nl}
            </span>
            <span
              className={`shrink-0 inline-flex items-center rounded-full px-2 py-0.5 text-[10px] font-semibold ${badge.className}`}
              data-source-type={row.source_type}
            >
              {badge.label}
            </span>
          </div>

          {/* Column 2: status pill */}
          <StatusPill status={row.status} errorMessage={row.error_message} />

          {/* Column 3: last_checked relative */}
          <span className="text-[11px] text-slate-500">
            {row.last_checked ? formatRelativeNL(row.last_checked) : 'Nog niet gecheckt'}
          </span>

          {/* Column 4: actions (ml-auto pushes right on md+) */}
          <div className="flex items-center gap-1 md:ml-auto">
            {row.available_actions.map((action) => (
              <ActionButton
                key={action.action}
                action={action}
                gemeenteCode={gemeenteCode}
                sourceKey={row.source_key}
                sourceLabel={row.label_nl}
                isRefreshing={isRefreshing && action.action === 'refresh'}
                onRefresh={() =>
                  refreshMutation.mutate({
                    gemeenteCode,
                    sourceKey: row.source_key,
                    sourceLabel: row.label_nl,
                  })
                }
              />
            ))}
          </div>
        </div>
      )
    }

    interface ActionButtonProps {
      action: Action
      gemeenteCode: string
      sourceKey: string
      sourceLabel: string
      isRefreshing: boolean
      onRefresh: () => void
    }

    function ActionButton({
      action,
      isRefreshing,
      onRefresh,
    }: ActionButtonProps) {
      const Icon = isRefreshing && action.action === 'refresh' ? Loader2 : ACTION_ICON[action.action]
      const isDisabled = !action.enabled || isRefreshing
      const baseClass =
        'inline-flex items-center gap-1 rounded-full px-3 py-1.5 text-[11px] font-semibold transition-colors'
      const enabledClass = 'bg-[#0E1111] text-white hover:opacity-90'
      const disabledClass = 'bg-slate-100 text-slate-400 cursor-not-allowed'

      // Upload is a deep-link to /instellingen/beleid-beheer per 03-CONTEXT.md §12
      if (action.action === 'upload' && action.enabled) {
        return (
          <Link
            to="/instellingen/beleid-beheer"
            className={`${baseClass} ${enabledClass}`}
            data-action={action.action}
            data-enabled="true"
          >
            <Icon size={14} strokeWidth={2} />
            <span>{action.label_nl}</span>
          </Link>
        )
      }

      const btn = (
        <button
          type="button"
          disabled={isDisabled}
          onClick={action.action === 'refresh' && action.enabled ? onRefresh : undefined}
          className={`${baseClass} ${isDisabled ? disabledClass : enabledClass}`}
          data-action={action.action}
          data-enabled={action.enabled ? 'true' : 'false'}
        >
          <Icon
            size={14}
            strokeWidth={2}
            className={isRefreshing ? 'animate-spin' : ''}
          />
          <span>{action.label_nl}</span>
        </button>
      )

      if (!action.enabled && action.reason_disabled) {
        return (
          <TooltipProvider delayDuration={150}>
            <Tooltip>
              <TooltipTrigger asChild>{btn}</TooltipTrigger>
              <TooltipContent side="top" className="max-w-xs text-[11px]">
                {action.reason_disabled}
              </TooltipContent>
            </Tooltip>
          </TooltipProvider>
        )
      }
      return btn
    }
    ```

    NOTE on TooltipTrigger asChild with a `disabled` button: Radix swallows pointer events on disabled buttons. The shadcn pattern works by wrapping in a span. If hovering a disabled button does not trigger the tooltip in manual smoke testing, wrap `btn` in `<span tabIndex={0}>{btn}</span>` inside TooltipTrigger. Storybook play in 03-04 will exercise this — if the play fails on tooltip queries, apply this wrap.

    ```tsx
    /**
     * GemeentePicker — M049/S03
     *
     * Horizontal pill-tab strip for switching active gemeente. Each pill shows:
     *   - gemeente_naam
     *   - tiny status-dot (groen all-live / amber any-missing / rood any-broken)
     *
     * Source: GemeenteSummary[] from useGemeentenSummary, sorted last_activity_at DESC.
     * Tokens locked in 03-CONTEXT.md §4. Motion layoutId="gemeente-picker" is SEPARATE
     * from SettingsTabNav's "settings-tab" — different animation scope.
     */
    import { motion } from 'framer-motion'
    import { Link } from '@tanstack/react-router'
    import type { GemeenteSummary } from '@/lib/types/data-sources-admin'

    interface Props {
      gemeenten: GemeenteSummary[]
      activeGemeenteCode: string
    }

    function dotColor(summary: GemeenteSummary): string {
      if ((summary.status_counts.broken ?? 0) > 0) return 'bg-red-500'
      if ((summary.status_counts.missing ?? 0) > 0) return 'bg-[#F9A826]'
      return 'bg-[#22C55E]'
    }

    export function GemeentePicker({ gemeenten, activeGemeenteCode }: Props) {
      return (
        <nav
          className="flex gap-1 overflow-x-auto rounded-full bg-[#F2F6F3] p-1 w-fit max-w-full"
          aria-label="Gemeente-keuze"
          data-component="gemeente-picker"
        >
          {gemeenten.map((g) => {
            const isActive = g.gemeente_code === activeGemeenteCode
            return (
              <Link
                key={g.gemeente_code}
                to="/instellingen/data-bronnen/$gemeenteCode"
                params={{ gemeenteCode: g.gemeente_code }}
                className="relative shrink-0 px-4 py-2 rounded-full text-[12px] font-semibold transition-colors"
                data-gemeente-code={g.gemeente_code}
                data-active={isActive ? 'true' : 'false'}
              >
                {isActive && (
                  <motion.div
                    layoutId="gemeente-picker"
                    className="absolute inset-0 rounded-full bg-[#0E1111]"
                    transition={{ type: 'spring', stiffness: 380, damping: 30 }}
                  />
                )}
                <span
                  className={`relative z-10 inline-flex items-center gap-1.5 ${
                    isActive ? 'text-white' : 'text-slate-600'
                  }`}
                >
                  <span className={`size-1.5 rounded-full ${dotColor(g)}`} aria-hidden />
                  {g.gemeente_naam}
                </span>
              </Link>
            )
          })}
        </nav>
      )
    }
    ```
