Required additions — 3 cases for BRK factors. Reference S03 factor outputs (from backend/app/services/score_v3_factors.py):

```typescript
case 'hypotheek_risico': {
  // backend waarde: boolean (true = heeft_hypotheek aanwezig op tenminste 1 eigenaar)
  const has = factor.waarde === true || factor.waarde === 'ja' || factor.waarde === 'Ja'
  return {
    value: has ? 'Ja' : 'Nee',
    sub: has ? 'Hypotheek-registratie aanwezig' : 'Geen hypotheek',
  }
}
case 'eigendoms_complexiteit': {
  // backend waarde: string ('enkelvoudig' | 'gedeeld' | 'meerdere_zakelijk_recht_types' etc.)
  const v = String(factor.waarde ?? '—')
  return { value: v.charAt(0).toUpperCase() + v.slice(1), sub: '' }
}
case 'koopwaarde_trend': {
  // backend waarde: dict {koopjaar, koopsom} OR int koopjaar fallback
  if (factor.waarde && typeof factor.waarde === 'object' && 'koopjaar' in (factor.waarde as object)) {
    const w = factor.waarde as { koopjaar: number | null; koopsom: number | null }
    const jaar = w.koopjaar !== null ? String(w.koopjaar) : '—'
    const sum = w.koopsom !== null
      ? new Intl.NumberFormat('nl-NL', { style: 'currency', currency: 'EUR', maximumFractionDigits: 0 }).format(w.koopsom)
      : ''
    return { value: jaar, sub: sum }
  }
  return { value: String(factor.waarde ?? '—'), sub: 'koopjaar' }
}
```
