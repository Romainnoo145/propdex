```typescript
// useBrkRecords.ts (~50 LOC)
import { useQuery } from '@tanstack/react-query'
import { apiClient } from '@/lib/api'
import type { BRKOwnershipRecordRead } from '@/lib/types/brk'

export function useBrkRecords(parcelId: string) {
  return useQuery<BRKOwnershipRecordRead[]>({
    queryKey: ['brk', parcelId],
    queryFn: async () => {
      const { data } = await apiClient.get<BRKOwnershipRecordRead[]>(
        `/api/parcels/${parcelId}/brk`
      )
      return data
    },
    enabled: Boolean(parcelId),
    staleTime: 30_000,
  })
}

// usePersistBrk.ts (~80 LOC)
export function usePersistBrk(parcelId: string) {
  const qc = useQueryClient()
  return useMutation<BRKOwnershipRecordRead[], AxiosError, void>({
    mutationFn: async () => {
      const { data } = await apiClient.post<BRKOwnershipRecordRead[]>(
        `/api/parcels/${parcelId}/brk/persist`
      )
      return data
    },
    onSuccess: (records) => {
      qc.setQueryData(['brk', parcelId], records)
      qc.invalidateQueries({ queryKey: ['scan', parcelId] })
    },
  })
  // Consumer reads error.response?.status + maps via mapBrkError util
}

// useRescorePoll.ts (~70 LOC)
export function useRescorePoll(opts: {
  parcelId: string
  pollStartAt: number  // unix ms when persist completed
  enabled: boolean
}): { phase: 'polling' | 'success' | 'timeout' | 'idle'; tickCount: number }
```

Note on 503 disambiguation: backend returns FeatureDisabled vs CredentialNotConfigured both as 503. Disambiguate via `detail` substring:
- detail.toLowerCase().includes('credential') → CredentialNotConfigured
- otherwise → FeatureDisabled

Polling cadence (verbatim AD3):
- Interval: 2000ms
- Max ticks: 30 (60s total window)
- Stop condition: flags.cascade_rescored_at (ISO date string) → Date.parse() > pollStartAt
- Cleanup: useEffect return clears interval
- No retry-loop on timeout
