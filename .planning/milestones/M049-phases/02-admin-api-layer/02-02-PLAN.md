<advisory_lock_pattern>
<!-- Verified per 02-RESEARCH.md §2 — use type_coerce(BigInteger) + blake2b stable hash -->

```python
import hashlib
from sqlalchemy import BigInteger, func, select, type_coerce


def compute_lock_key(gemeente_code: str, source_key: str) -> int:
    """Cross-process stable 63-bit lock key (PYTHONHASHSEED-safe)."""
    raw = f"{gemeente_code}:{source_key}".encode()
    digest = hashlib.blake2b(raw, digest_size=8).digest()
    return int.from_bytes(digest, "big", signed=False) & 0x7FFFFFFFFFFFFFFF


# Inside route handler:
lock_acquired: bool = await db.scalar(
    select(func.pg_try_advisory_xact_lock(type_coerce(compute_lock_key(code, key), BigInteger())))
)
# If False → another request is in-flight; return current row, no probe run.
```

**Why type_coerce + BigInteger:** asyncpg defaults Python int to int4; hash values >2^31 overflow. type_coerce forces int8. (Research §2 pitfall 2.)

**Why blake2b not hash():** Python's hash() is PYTHONHASHSEED-randomized → multiple workers compute different keys → no cross-worker contention protection. (Research §pitfall 6.)
</advisory_lock_pattern>

    ```python
    async def _load_or_create_row(
        gemeente_code: str, source_key: str, db: AsyncSession,
    ) -> DataSourceStatus:
        """Get existing data_source_status row, or create a transient (not yet committed) one."""
        existing = await db.scalar(
            select(DataSourceStatus)
            .where(DataSourceStatus.gemeente_code == gemeente_code)
            .where(DataSourceStatus.source_key == source_key)
        )
        if existing is not None:
            return existing
        row = DataSourceStatus(
            id=uuid.uuid4(),
            gemeente_code=gemeente_code,
            source_key=source_key,
            status="missing",
            last_checked=None,
            error_message=None,
            configuratie=None,
        )
        db.add(row)
        await db.flush()
        return row


    @router.post("/{gemeente_code}/{source_key}/refresh", response_model=DataSourceRow)
    async def refresh_source(
        gemeente_code: GemeenteCodePath,
        source_key: str,
        _: User = Depends(get_current_superuser),
        db: AsyncSession = Depends(get_db),
    ) -> DataSourceRow:
        """Refresh status for one (gemeente_code, source_key) tuple.

        Idempotent: advisory lock guards concurrent calls; compare-and-update means
        unchanged probe results do not bump last_checked.
        """
        entry = get_entry(source_key)
        if entry is None:
            raise HTTPException(status_code=404, detail="Source key not in catalog")

        # Advisory lock (auto-released on transaction commit/rollback)
        lock_key = compute_lock_key(gemeente_code, source_key)
        lock_acquired: bool = await db.scalar(
            select(func.pg_try_advisory_xact_lock(type_coerce(lock_key, BigInteger())))
        )

        row = await _load_or_create_row(gemeente_code, source_key, db)

        if not lock_acquired:
            # Another request is mid-refresh — return current state, no probe run, no write.
            return _row_from_db(row, entry)

        # Dispatch to probe by source_type
        probe_fn = PROBE_REGISTRY[entry.source_type]
        probe_result = await probe_fn(source_key, gemeente_code, db)

        # Compare-and-update: write only if status or error_message changed
        if row.status != probe_result["status"] or row.error_message != probe_result["error_message"]:
            row.status = probe_result["status"]
            row.error_message = probe_result["error_message"]
            row.last_checked = datetime.utcnow()
            await db.commit()
        else:
            # No change → no write. Ensure transient (newly-created) row is still persisted.
            # If row was just created and probe also returned "missing", we still want to keep it.
            # `flush` already persisted it; commit so it survives the request.
            await db.commit()

        return _row_from_db(row, entry)
    ```
