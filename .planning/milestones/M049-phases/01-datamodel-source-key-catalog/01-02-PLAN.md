    ```python
    """Alembic up+down smoke test for M049/S01 data_source_status migration.

    Runs against propdex_test via Python API (not subprocess).
    """
    import os
    import pytest
    from alembic import command
    from alembic.config import Config
    from sqlalchemy import inspect, text
    from sqlalchemy.ext.asyncio import create_async_engine


    @pytest.mark.asyncio
    async def test_upgrade_then_downgrade_clean():
        # NOTE: project alembic.ini lives in backend/; resolve from this file's dir.
        from pathlib import Path
        backend_root = Path(__file__).resolve().parents[2]
        alembic_cfg = Config(str(backend_root / "alembic.ini"))

        # Determine the M049 revision SHA dynamically — look for file matching pattern.
        versions_dir = backend_root / "alembic" / "versions"
        m049_files = list(versions_dir.glob("z0_m049_s01_*.py"))
        assert m049_files, "Migration file z0_m049_s01_* not found"
        m049_revision_file = m049_files[0].read_text()
        import re
        m = re.search(r"revision:\s*str\s*=\s*['\"]([^'\"]+)['\"]", m049_revision_file)
        assert m, "Could not parse revision SHA from migration"
        target_revision = m.group(1)

        # Upgrade to head (which is now this revision)
        command.upgrade(alembic_cfg, "head")

        # Verify table exists
        engine = create_async_engine(os.environ["DATABASE_URL"])
        async with engine.connect() as conn:
            result = await conn.run_sync(
                lambda sync_conn: inspect(sync_conn).has_table("data_source_status")
            )
            assert result, "data_source_status table missing after upgrade"

            cols = await conn.run_sync(
                lambda sync_conn: {c["name"] for c in inspect(sync_conn).get_columns("data_source_status")}
            )
            expected = {
                "id", "gemeente_code", "source_key", "status", "last_checked",
                "error_message", "configuratie", "managed_by_user_id",
                "created_at", "updated_at",
            }
            assert expected.issubset(cols), f"Missing columns: {expected - cols}"

            # Verify UQ exists
            uqs = await conn.run_sync(
                lambda sync_conn: {
                    uq["name"] for uq in inspect(sync_conn).get_unique_constraints("data_source_status")
                }
            )
            assert any("gemeente_source" in n or "gemeente_code" in n for n in uqs), (
                f"UQ on (gemeente_code, source_key) missing. Found UQs: {uqs}"
            )

        # Downgrade one step — table should disappear
        command.downgrade(alembic_cfg, "-1")
        async with engine.connect() as conn:
            still_exists = await conn.run_sync(
                lambda sync_conn: inspect(sync_conn).has_table("data_source_status")
            )
            assert not still_exists, "data_source_status table NOT dropped by downgrade"

        # Restore to head for other tests
        command.upgrade(alembic_cfg, "head")
        await engine.dispose()
    ```

    ```python
    """DataSourceStatus model — vendor-managed catalog row per (gemeente, source_key) (M049/S01).

    DELIBERATELY OMITS TenantMixin: Propdex BV (superuser) maintains this catalog
    centrally for all customer orgs. RBAC is enforced at the route layer via
    Depends(get_current_superuser), NOT at the row layer. Eng-review locked 2026-05-14.
    """

    import uuid
    from datetime import datetime
    from typing import Literal

    from pydantic import BaseModel, ConfigDict
    from sqlalchemy import ForeignKey, Index, String, Text, UniqueConstraint, text
    from sqlalchemy.dialects.postgresql import JSONB, TIMESTAMP
    from sqlalchemy.orm import Mapped, mapped_column

    from app.models.base import Base, TimestampMixin

    DataSourceStatusValue = Literal["live", "broken", "missing", "not_applicable"]


    class DataSourceStatus(Base, TimestampMixin):
        """Vendor-managed status row for one (gemeente_code, source_key) combination.

        Status enum is validated at the Pydantic layer via `DataSourceStatusCreate`.
        DB column is plain String(20) — no Postgres native ENUM (eng-review: avoid
        migration-friction; revisit when statuses stabilize).
        """

        __tablename__ = "data_source_status"

        id: Mapped[uuid.UUID] = mapped_column(
            primary_key=True,
            server_default=text("gen_random_uuid()"),
        )
        gemeente_code: Mapped[str] = mapped_column(
            String(6),
            nullable=False,
            comment="Gemeente identifier (e.g. 'GM0983' for Venlo)",
        )
        source_key: Mapped[str] = mapped_column(
            String(64),
            nullable=False,
            comment="Kebab-case key from data_source_keys catalog (e.g. 'woonvisie')",
        )
        status: Mapped[str] = mapped_column(
            String(20),
            nullable=False,
            comment="One of: live, broken, missing, not_applicable. Validated via Pydantic.",
        )
        last_checked: Mapped[datetime | None] = mapped_column(
            TIMESTAMP(timezone=False),
            nullable=True,
            comment="When status was last confirmed by probe / refresh",
        )
        error_message: Mapped[str | None] = mapped_column(
            Text,
            nullable=True,
            comment="Diagnostic message if status=broken",
        )
        configuratie: Mapped[dict | None] = mapped_column(
            JSONB,
            nullable=True,
            comment="Non-secret state (refresh-interval, last_synced_doc_id, retry-count). "
                    "Secrets live in integration_credentials (Phase 5).",
        )
        managed_by_user_id: Mapped[uuid.UUID | None] = mapped_column(
            ForeignKey("users.id", ondelete="SET NULL"),
            nullable=True,
            comment="Vendor superuser who last managed this row (audit-trail)",
        )

        __table_args__ = (
            UniqueConstraint(
                "gemeente_code",
                "source_key",
                name="uq_data_source_status_gemeente_source",
            ),
            Index("ix_data_source_status_gemeente_code", "gemeente_code"),
        )


    # ─── Pydantic schemas (status-enum validation) ──────────────────────


    class DataSourceStatusCreate(BaseModel):
        """Create-payload — enforces status enum at Pydantic layer."""

        model_config = ConfigDict(extra="forbid")

        gemeente_code: str
        source_key: str
        status: DataSourceStatusValue
        last_checked: datetime | None = None
        error_message: str | None = None
        configuratie: dict | None = None
        managed_by_user_id: uuid.UUID | None = None


    class DataSourceStatusRead(BaseModel):
        """Read-payload returned by GET endpoints (Phase 2)."""

        model_config = ConfigDict(from_attributes=True)

        id: uuid.UUID
        gemeente_code: str
        source_key: str
        status: DataSourceStatusValue
    ```
