     * M048 Phase 7 — Factor groups (bron-mapping) for UI pagination.
     *
     * MIRROR of backend FACTOR_REGISTRY[key].bron values from
     * backend/app/services/score_v3.py (Phase 7 Plan 03 Task 1).
     * Keep these in sync if backend mapping changes.
     *
     * UI-only: capaciteit (derived UI card, not in backend registry) is
     * mapped to 'fysiek' here. Document this exception below.

    export type Bron =
      | 'gemeente_beleid'
      | 'markt'
      | 'fysiek'
      | 'klimaat'
      | 'juridisch'
      | 'planologisch'

    /** Display order for group pages (left-to-right in pagination). */
    export const BRON_ORDER: readonly Bron[] = [
      'gemeente_beleid',
      'markt',
      'fysiek',
      'klimaat',
      'juridisch',
      'planologisch',
    ] as const

      gemeente_beleid: 'GEMEENTE BELEID',
      markt: 'MARKT',

      klimaat: 'KLIMAAT',

      juridisch_klimaat: 'juridisch',
      criminaliteit: 'markt',
      max_hoogte_bereikt: 'fysiek',
      natura2000: 'klimaat',
      overstromingsrisico: 'klimaat',
      geluid: 'klimaat',
      luchtkwaliteit: 'klimaat',
    }

    /** Card shape consumed by buildPages — minimal contract. */
    export interface PageCard {
      label: string
      value: string
      sub: string
      pts: number
      factor_key: string
      bron: Bron
      missing?: boolean
    }

    /** Pin shape needed by buildPages. Mirrors useFactorPins FactorPin. */
    export interface PinLike {
      id: string
      factor_key: string
      position: number
    }

    /** Synthetic ghost slot rendered when pinned-page is partial. */
    export type GhostSlot = { kind: 'ghost' }

    export type PageSlot = PageCard | GhostSlot

    export interface BuiltPage {
      title: string
      kind: 'pinned' | 'group'
      cards: PageSlot[]
    }

    export const CARDS_PER_PAGE = 6

    /**
     * Build the full page array: page 0 = pinned synth-page; page 1..N = group pages.
     *
     * Page 0 logic:
     *   - 0 pins → title "TOP FACTOREN", show top-6 by |pts| desc across all bronnen
     *   - 1..5 pins → title "GEPIND", show pinned cards in position order + ghost slots filling to 6
     *   - 6 pins → title "GEPIND", show 6 pinned cards (no ghosts)
     *
     * Group pages (one per bron in BRON_ORDER, skipping empty):
     *   - title = BRON_LABEL[bron]
     *   - cards = allCards filtered to that bron, sliced to CARDS_PER_PAGE
     *   - cards keep their `factor_key` so pinned ones can render the ⭐
     */
    export function buildPages(
      allCards: PageCard[],
      pins: PinLike[],
    ): BuiltPage[] {
      // Page 0: pinned or fallback
      let pinnedPage: BuiltPage
      if (pins.length === 0) {
        const top6 = [...allCards]
          .sort((a, b) => Math.abs(b.pts) - Math.abs(a.pts))
          .slice(0, CARDS_PER_PAGE)
        pinnedPage = {
          title: 'TOP FACTOREN',
          kind: 'pinned',
          cards: top6,
        }
      } else {
        const ordered = [...pins].sort((a, b) => a.position - b.position)
        const pinnedCards: PageSlot[] = ordered.map((p) => {
          const card = allCards.find((c) => c.factor_key === p.factor_key)
          // If pinned card no longer in allCards, render as missing ghost-shaped placeholder
          if (!card) {
            return {
              label: p.factor_key.toUpperCase(),
              value: '—',
              sub: 'Geen data',
              pts: 0,
              factor_key: p.factor_key,
              bron: FACTOR_BRON[p.factor_key] ?? 'fysiek',
              missing: true,
            } as PageCard
          }
          return card
        })
        const ghostsNeeded = Math.max(0, CARDS_PER_PAGE - pinnedCards.length)
        const cards: PageSlot[] = [
          ...pinnedCards,
          ...Array.from({ length: ghostsNeeded }, () => ({ kind: 'ghost' as const })),
        ]
        pinnedPage = { title: 'GEPIND', kind: 'pinned', cards }
      }

      // Group pages
      const groupPages: BuiltPage[] = BRON_ORDER.map((bron) => {
        const cards = allCards
          .filter((c) => c.bron === bron)
          .slice(0, CARDS_PER_PAGE)
        return { title: BRON_LABEL[bron], kind: 'group' as const, cards }
      }).filter((p) => p.cards.length > 0)

      return [pinnedPage, ...groupPages]
    }

    **File 2: frontend/src/hooks/useFactorPins.ts** (NEW)

    ```ts
    /**
     * M048 Phase 7 — useFactorPins hook (PIN-01, PIN-03, UI-01).
     *
     * TanStack Query hook for per-user factor pins with optimistic updates.
     * Mirror shape of useScoreWeights.ts (project canonical pattern).
     *
     * Backend contract (Plan 02):
     *   GET    /api/users/me/factor-pins?category={potentie|risico} → 200 FactorPin[]
     *   POST   /api/users/me/factor-pins {category, factor_key}     → 201 FactorPin | 409 | 400 MAX_PINS_REACHED
     *   DELETE /api/users/me/factor-pins/{id}                       → 204
     *
     * Max-6 guard fires `toast("⭐ Max 6 bereikt")` BEFORE POST when adding
     * would exceed 6 pins; server 400 is silent backup-net.
     */

    import { useQuery, useMutation, useQueryClient } from '@tanstack/react-query'
    import { toast } from 'sonner'

    import { useAuth } from '@/lib/auth/useAuth'
    import { apiFetch } from '@/lib/api'

    export type PinCategory = 'potentie' | 'risico'

    export interface FactorPin {
      id: string
      category: PinCategory
      factor_key: string
      position: number
      created_at: string
    }

    export const MAX_PINS_PER_CATEGORY = 6

    const BASE = '/api/users/me/factor-pins'

    interface UseFactorPinsResult {
      pins: FactorPin[]
      isLoading: boolean
      toggle: (factorKey: string) => void
      isPinned: (factorKey: string) => boolean
    }

    export function useFactorPins(category: PinCategory): UseFactorPinsResult {
      const { accessToken } = useAuth()
      const qc = useQueryClient()
      const queryKey = ['factor-pins', category] as const

      const query = useQuery<FactorPin[], Error>({
        queryKey: [...queryKey],
        queryFn: () =>
          apiFetch<FactorPin[]>(`${BASE}?category=${category}`, {
            accessToken,
            method: 'GET',
          }),
        enabled: !!accessToken,
        staleTime: 5 * 60 * 1000, // 5 min
      })

      const addMutation = useMutation<
        FactorPin,
        Error,
        string,
        { previous: FactorPin[] }
      >({
        mutationFn: (factor_key) =>
          apiFetch<FactorPin>(BASE, {
            accessToken,
            method: 'POST',
            body: JSON.stringify({ category, factor_key }),
          }),
        onMutate: async (factor_key) => {
          await qc.cancelQueries({ queryKey: [...queryKey] })
          const previous = qc.getQueryData<FactorPin[]>([...queryKey]) ?? []
          const nextPosition =
            previous.length > 0
              ? (previous[previous.length - 1]?.position ?? -1) + 1
              : 0
          const optimistic: FactorPin = {
            id: `tmp-${factor_key}`,
            category,
            factor_key,
            position: nextPosition,
            created_at: new Date().toISOString(),
          }
          qc.setQueryData<FactorPin[]>([...queryKey], [...previous, optimistic])
          return { previous }
        },
        onError: (_err, _key, ctx) => {
          if (ctx) qc.setQueryData([...queryKey], ctx.previous)
        },
        onSettled: () => {
          qc.invalidateQueries({ queryKey: [...queryKey] })
        },
      })

      const removeMutation = useMutation<
        void,
        Error,
        string,
        { previous: FactorPin[] }
      >({
        mutationFn: (pinId) =>
          apiFetch<void>(`${BASE}/${pinId}`, {
            accessToken,
            method: 'DELETE',
          }),
        onMutate: async (pinId) => {
          await qc.cancelQueries({ queryKey: [...queryKey] })
          const previous = qc.getQueryData<FactorPin[]>([...queryKey]) ?? []
          qc.setQueryData<FactorPin[]>(
            [...queryKey],
            previous.filter((p) => p.id !== pinId),
          )
          return { previous }
        },
        onError: (_err, _id, ctx) => {
          if (ctx) qc.setQueryData([...queryKey], ctx.previous)
        },
        onSettled: () => {
          qc.invalidateQueries({ queryKey: [...queryKey] })
        },
      })

      const pins = query.data ?? []

      const isPinned = (factorKey: string): boolean =>
        pins.some((p) => p.factor_key === factorKey)

      const toggle = (factorKey: string): void => {
        const existing = pins.find((p) => p.factor_key === factorKey)
        if (existing) {
          removeMutation.mutate(existing.id)
          return
        }
        if (pins.length >= MAX_PINS_PER_CATEGORY) {
          toast('⭐ Max 6 bereikt')
          return
        }
        addMutation.mutate(factorKey)
      }

      return {
        pins,
        isLoading: query.isLoading,
        toggle,
        isPinned,
      }
    }
    ```
