ContactPhaseView dispatcher logic:
```typescript
export function ContactPhaseView({ parcelId }: { parcelId: string }) {
  const { data: records, isLoading } = useBrkRecords(parcelId)
  const [pollStartAt, setPollStartAt] = useState<number | null>(null)

  const onPersistSuccess = () => setPollStartAt(Date.now())

  if (isLoading) return <SkeletonState />
  const hasRecords = (records?.length ?? 0) > 0

  return (
    <ResultsShell>
      {/* Back-to-fase-1 link */}
      {/* Persist panel (always rendered — handles CTA + error + empty states) */}
      <BrkPersistPanel parcelId={parcelId} records={records ?? []} onPersistSuccess={onPersistSuccess} />
      {/* Record display when present */}
      {hasRecords && (
        <>
          <BrkOwnerCard records={records!} />
          <BrkHypotheekCard records={records!} />
          <BrkKoopwaardeTimeline records={records!} />
        </>
      )}
      {/* Rescore banner — only after persist mutation has fired AND records non-empty */}
      {pollStartAt !== null && hasRecords && (
        <RescoreProgressBanner parcelId={parcelId} pollStartAt={pollStartAt} />
      )}
    </ResultsShell>
  )
}
```

BrkPersistPanel state machine:
```
records.length === 0 && !mutation.isPending && !mutation.error && !mutation.isSuccess → IDLE (show CTA "Vraag BRK eigendom op")
mutation.isPending → LOADING (button disabled, spinner)
mutation.error → ERROR (mapBrkError(error.response?.status, error.response?.data?.detail) → render title/body, retry button if retryable)
mutation.isSuccess && records.length === 0 → EMPTY (AD4 "Particuliere eigenaar" infoblok)
records.length > 0 → SUCCESS (small confirmation, no CTA — main display in BrkOwnerCard etc.)
```
