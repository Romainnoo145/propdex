```python
    current_weights: dict[str, int],   # {factor_key: pre_weight} for 13 existing risico
    target_sum: int = 80,
) -> dict[str, int]:
    """Integer rebalance via largest-remainder method. Deterministic.

    Args:
        current_weights: existing risico factor weights summing to 100.
        target_sum: post-rebalance sum (80 — leaves 20 for 3 new BRK factors).

    Returns:
        new_weights dict, summing to target_sum, integer-valued.

    Algorithm:
        1. raw = {k: w * target_sum / 100 for k, w in current_weights.items()}
        2. floor_weights = {k: int(raw[k]) for k in raw}

        4. shortfall = target_sum - sum(floor_weights.values())
        5. Sort keys by (-remainder, key) so largest remainder wins, ties broken

        6. Allocate +1 to the top `shortfall` keys.
        7. Assert sum(new_weights.values()) == target_sum.
    """
```

```python
def score_hypotheek_risico(parcel_data: dict) -> FactorResult:
    """Factor R14: Hypotheek-risico. Default gewicht=10 (D2 + D4).

    Empty BRK records → missing=True via _missing_brk (D3).
    Multi-owner parcels: ANY mortgaged owner flags risico (binary on any-True).
    """
    records = parcel_data.get("brk_records") or []
    if not records:
        return _missing_brk(
            "Geen BRK-data — eigendomspositie onbekend",
            weight=10,
        )
    n_with = sum(1 for r in records if r.get("heeft_hypotheek"))
    n_total = len(records)
    score = 1.0 if n_with > 0 else 0.0
    return FactorResult(
        waarde={"n_with_hypotheek": n_with, "n_total": n_total},
        genormaliseerde_score=score,
        gewicht=10,
        bijdrage=score * 10,
        categorie="risico",
        toelichting_kort=(
            f"{n_with}/{n_total} eigenaren met hypotheek" if n_with else "Geen hypotheek geregistreerd"
        ),
        toelichting_lang=(
            f"BRK toont {n_with} van {n_total} eigenaren met hypotheek-vermelding. "
            "Onderhandeling vereist mogelijk hypotheek-vrije levering."
            if n_with else
            f"Geen van de {n_total} eigenaren heeft een hypotheek geregistreerd."
        ),
        missing=False,
        data_bronnen=["brk_ownership_records"],
    )


def score_eigendoms_complexiteit(parcel_data: dict) -> FactorResult:
    """Factor R15: Eigendoms-complexiteit. Default gewicht=6 (D2 + D5).

    D5 (user-locked): count distinct (eigenaar_naam, kvk_nummer). Ignores
    aandeel + zakelijk_recht_type. Score: 1→0.0, 2→0.5, ≥3→1.0.
    """
    records = parcel_data.get("brk_records") or []
    if not records:
        return _missing_brk(
            "Geen BRK-data — eigendomspositie onbekend",
            weight=6,
        )
    unique_owners = {(r.get("eigenaar_naam"), r.get("kvk_nummer")) for r in records}
    n_owners = len(unique_owners)
    if n_owners <= 1:
        score = 0.0
    elif n_owners == 2:
        score = 0.5
    else:
        score = 1.0
    return FactorResult(
        waarde={"n_owners": n_owners},
        genormaliseerde_score=score,
        gewicht=6,
        bijdrage=score * 6,
        categorie="risico",
        toelichting_kort=f"{n_owners} verschillende eigenaren",
        toelichting_lang=(
            f"BRK toont {n_owners} unieke eigenaren. "
            f"{'Eén eigenaar — eenvoudige onderhandelingspositie.' if n_owners == 1 else 'Multi-owner deal vereist coördinatie met meerdere partijen.'}"
        ),
        missing=False,
        data_bronnen=["brk_ownership_records"],
    )


def score_koopwaarde_trend(parcel_data: dict) -> FactorResult:
    """Factor R16: Koopsom-actualiteit (NOT 'trend' — D6 narrative honesty rule).

    Default gewicht=4 (D2 + D6). koopjaar-age proxy from most-recent record.
    Buckets: <5y → 0.3, 5–15y → 0.5, 15–25y → 0.7, >25y → 0.9.
    NULL koopjaar OR future koopjaar OR no records → missing=True (F5 chosen behaviour).
    """
    from datetime import datetime
    records = parcel_data.get("brk_records") or []
    if not records:
        return _missing_brk("Koopjaar onbekend — geen BRK-data", weight=4)
    most_recent = max(records, key=lambda r: r.get("fetched_at") or "")
    koopjaar = most_recent.get("koopjaar")
    if koopjaar is None:
        return _missing_brk("Geen koopjaar geregistreerd", weight=4)
    current_year = datetime.utcnow().year
    age = current_year - int(koopjaar)
    if age < 0:  # F5: future koopjaar = data error
        return _missing_brk("Ongeldige koopjaar-data", weight=4)
    if age < 5:
        score, qualifier = 0.3, "recente koop"
    elif age < 15:
        score, qualifier = 0.5, "actuele koopsom-vermelding"
    elif age < 25:
        score, qualifier = 0.7, "koopsom mogelijk verouderd"
    else:
        score, qualifier = 0.9, "koopsom waarschijnlijk sterk verouderd"
    return FactorResult(
        waarde={"koopjaar": koopjaar, "age_jaar": age, "koopsom": most_recent.get("koopsom")},
        genormaliseerde_score=score,
        gewicht=4,
        bijdrage=score * 4,
        categorie="risico",
        toelichting_kort=f"Koop in {koopjaar} ({age}j geleden) — {qualifier}",
        toelichting_lang=(
            f"Meest recente BRK-record toont koop in {koopjaar} ({age} jaar geleden). "
            "Dit is een actualiteits-proxy van de koopsom, geen waarde-ontwikkeling over tijd."
        ),
        missing=False,
        data_bronnen=["brk_ownership_records"],
    )
```

    ```python
    def _rebalance_risico_weights(
        current_weights: dict[str, int],
        target_sum: int = 80,
    ) -> dict[str, int]:
        """Integer rebalance via largest-remainder method (M050/S03 D2).

        Used during one-time pre-S03 → post-S03 registry migration to scale 13 existing
        risico-factor weights from sum=100 to sum=80, freeing 20 weight-units for the
        3 new BRK risico-factors. Public for reuse by future weight calibrations
        (M050/S03 Carry-2).

        Algorithm:
            1. raw = {k: w * target_sum / 100 for k, w in current_weights.items()}
            2. floor_weights = {k: int(raw[k]) for k in raw}
            3. remainders = {k: raw[k] - floor_weights[k] for k in raw}
            4. shortfall = target_sum - sum(floor_weights.values())
            5. Sort keys by (-remainder, key) — largest remainder wins, ties broken
               alphabetically for determinism.
            6. Allocate +1 to the top `shortfall` keys.

        Args:
            current_weights: existing factor weights summing to 100.
            target_sum: post-rebalance sum (80 for S03).

        Returns:
            new_weights dict, integer-valued, summing to target_sum.

        Raises:
            AssertionError: if input weights do not sum to 100 OR
                output weights do not sum to target_sum (F6 invariant).
        """
        input_sum = sum(current_weights.values())
        assert input_sum == 100, (
            f"_rebalance_risico_weights input must sum to 100, got {input_sum}"
        )
        if not current_weights:
            return {}
        raw = {k: w * target_sum / 100 for k, w in current_weights.items()}
        floor_weights = {k: int(v) for k, v in raw.items()}
        remainders = {k: raw[k] - floor_weights[k] for k in raw}
        shortfall = target_sum - sum(floor_weights.values())
        # Sort by (-remainder, key) so largest remainder wins; alphabetical tie-break
        sorted_keys = sorted(remainders.keys(), key=lambda k: (-remainders[k], k))
        new_weights = dict(floor_weights)
        for k in sorted_keys[:shortfall]:
            new_weights[k] = new_weights[k] + 1
        output_sum = sum(new_weights.values())
        assert output_sum == target_sum, (
            f"_rebalance_risico_weights output sum {output_sum} != {target_sum} (F6)"
        )
        return new_weights
    ```
