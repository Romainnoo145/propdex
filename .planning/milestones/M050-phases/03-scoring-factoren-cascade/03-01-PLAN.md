    ```python
    async def enrich_parcel_data_with_brk(
        parcel_data: dict,
        org_id: UUID,
        db: AsyncSession,
    ) -> dict:
        """Materialize latest BRK rows for (org_id, parcel_id) into parcel_data['brk_records'].

        D1: single source of BRK data for scoring. Factor functions stay PURE — no DB calls.
        D8: when settings.M050_S03_FACTORS_ENABLED=False → returns parcel_data UNCHANGED.
             The 'brk_records' key is NOT inserted, so factor functions see None via
             parcel_data.get('brk_records') and return missing=True via _missing_brk.
        D9: read-only; never triggers HTTP fetch; never spends a Kadaster credit.

        Args:
            parcel_data: scan-job/rescore-job parcel context dict. MUST contain 'parcel_id'.
            org_id: caller organization (multi-tenant guard — REGRESSION-CRITICAL F2).
            db: async session bound to scoring transaction.

        Returns:
            parcel_data dict with 'brk_records': list[BRKRecordDict] populated (may be []).
            Same dict identity is returned (mutated in place for caller-side ergonomics).
        """
        if not settings.M050_S03_FACTORS_ENABLED:
            return parcel_data

        parcel_id = parcel_data.get("parcel_id")
        if parcel_id is None:
            logger.debug(
                "enrich_parcel_data_with_brk skipped: parcel_data missing 'parcel_id'"
            )
            parcel_data["brk_records"] = []
            return parcel_data

        stmt = (
            select(BrkOwnershipRecord)
            .where(
                BrkOwnershipRecord.organization_id == org_id,
                BrkOwnershipRecord.parcel_id == parcel_id,
                BrkOwnershipRecord.soft_deleted_at.is_(None),
            )
            .order_by(BrkOwnershipRecord.fetched_at.desc())
        )
        result = await db.execute(stmt)
        rows = result.scalars().all()

        records: list[BRKRecordDict] = []
        for row in rows:
            records.append(
                BRKRecordDict(
                    eigenaar_type=row.eigenaar_type,
                    eigenaar_naam=row.eigenaar_naam,
                    kvk_nummer=row.kvk_nummer,
                    statutaire_zetel=row.statutaire_zetel,
                    zakelijk_recht_type=row.zakelijk_recht_type,
                    aandeel=row.aandeel,
                    koopsom=float(row.koopsom) if row.koopsom is not None else None,
                    koopjaar=row.koopjaar,
                    heeft_hypotheek=bool(row.heeft_hypotheek),
                    fetched_at=row.fetched_at.isoformat(),
                )
            )

        parcel_data["brk_records"] = records
        logger.debug(
            "enrich_parcel_data_with_brk org=%s parcel=%s rows=%d",
            org_id, parcel_id, len(records),
        )
        return parcel_data
    ```
